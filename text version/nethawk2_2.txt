class RPCAPCaptureThread(QThread):
    """QThread to safely read packets from RPCAP stream (Wireshark-compatible)"""
    packet_received = pyqtSignal(dict)
    error_occurred = pyqtSignal(str)
    status_message = pyqtSignal(str)
    
    def __init__(self, socket_conn, rpcap_client, callback=None):
        super().__init__()
        self.socket = socket_conn
        self.rpcap_client = rpcap_client
        self.callback = callback
        self.running = False
        self.packet_counter = 0
    
    def run(self):
        """Read packets from RPCAP stream - Wireshark-compatible protocol"""
        self.running = True
        try:
            import struct
            
            self.status_message.emit("RPCAP capture started...")
            
            # Set socket timeout for packet reading (non-blocking with timeout)
            # This prevents indefinite blocking if no packets arrive
            self.socket.settimeout(1.0)  # 1 second timeout for recv operations
            
            while self.running and self.rpcap_client.connected:
                try:
                    # Read message header (8 bytes: RPCAP protocol header)
                    # Format: !BBHI (version:1, type:1, value:2, plen:4)
                    header = self.socket.recv(8)
                    if len(header) < 8:
                        if len(header) == 0:
                            # EOF - normal termination
                            LOGGER.info("RPCAP connection closed by server (EOF)")
                            break
                        self.error_occurred.emit(f"Incomplete RPCAP header: {len(header)}/8")
                        LOGGER.warning(f"Incomplete RPCAP header received: {len(header)}/8 bytes")
                        break
                    
                    ver, msg_type, value, msg_length = struct.unpack('!BBHI', header)
                    
                    LOGGER.debug(f"RPCAP message received: ver={ver}, type={msg_type}, value={value}, plen={msg_length}")
                    
                    if msg_type == self.rpcap_client.RPCAP_MSG_PACKET:
                        LOGGER.debug(f"RPCAP PACKET message (plen={msg_length})")
                        # Read packet header (20 bytes: RPCAP packet header format)
                        # Format: timestamp_sec (4) + timestamp_usec (4) + caplen (4) + len (4) + npkt (4)
                        pkt_header = self.socket.recv(20)
                        if len(pkt_header) < 20:
                            self.error_occurred.emit(f"Incomplete packet header: {len(pkt_header)}/20")
                            break
                        
                        # Parse packet header (big-endian for RPCAP)
                        timestamp_sec = struct.unpack('!I', pkt_header[0:4])[0]
                        timestamp_usec = struct.unpack('!I', pkt_header[4:8])[0]
                        caplen = struct.unpack('!I', pkt_header[8:12])[0]
                        len_pkt = struct.unpack('!I', pkt_header[12:16])[0]
                        npkt = struct.unpack('!I', pkt_header[16:20])[0]  # Packet number
                        
                        timestamp = timestamp_sec + timestamp_usec / 1000000.0
                        
                        # Validate packet length
                        if caplen > 65535 or caplen == 0:
                            self.error_occurred.emit(f"Invalid packet length: {caplen}")
                            break
                        
                        # Read packet data
                        packet_data = b''
                        while len(packet_data) < caplen:
                            chunk = self.socket.recv(min(caplen - len(packet_data), 65535))
                            if not chunk:
                                break
                            packet_data += chunk
                        
                        if len(packet_data) == caplen:
                            # Parse packet with scapy (Wireshark-compatible)
                            try:
                                scapy_packet = Ether(packet_data)
                                packet_info = parse_packet_from_scapy(
                                    scapy_packet,
                                    timestamp=timestamp,
                                    sip_ports=DEFAULT_SIP_PORTS,
                                    rtp_port_range=DEFAULT_RTP_PORT_RANGE
                                )
                                
                                if not packet_info:
                                    packet_info = {
                                        'timestamp': _format_packet_timestamp(timestamp),
                                        'src': '',
                                        'dst': '',
                                        'sport': 0,
                                        'dport': 0,
                                        'protocol': 'Unknown',
                                        'length': len_pkt,
                                        'qos_dscp': 0,
                                        'qos_name': 'BE',
                                        'flags': '',
                                        'info': 'RPCAP packet',
                                        'raw_packet': packet_data
                                    }
                                
                                self.packet_counter += 1
                                packet_info['no'] = self.packet_counter
                                packet_info['frame_number'] = self.packet_counter
                                packet_info['length'] = len_pkt
                                
                                # Emit packet safely via Qt signal (thread-safe)
                                self.packet_received.emit(packet_info)
                                
                                if self.packet_counter == 1:
                                    LOGGER.info(f"First RPCAP packet received and emitted (packet #{self.packet_counter})")
                                
                                if self.packet_counter % 100 == 0:
                                    LOGGER.info(f"RPCAP: Captured {self.packet_counter} packets")
                                    self.status_message.emit(f"Captured {self.packet_counter} packets")
                                    
                            except Exception as parse_error:
                                LOGGER.debug("RPCAP packet parse error: %s", parse_error, exc_info=True)
                                continue
                        else:
                            self.error_occurred.emit(f"Incomplete packet: {len(packet_data)}/{caplen}")
                            break
                            
                    elif msg_type == self.rpcap_client.RPCAP_MSG_ERROR:
                        # Error message payload (plen bytes)
                        if msg_length > 0:
                            error_data = self.socket.recv(msg_length)
                            error_msg = error_data.decode('utf-8', errors='ignore')
                            self.error_occurred.emit(f"RPCAP error: {error_msg}")
                        break
                    else:
                        # Unknown message type, skip payload
                        if msg_length > 0:
                            self.socket.recv(msg_length)
                            
                except socket.timeout:
                    # Timeout is normal when waiting for packets - just continue
                    # This allows the loop to check self.running periodically
                    continue
                except Exception as e:
                    if self.running:
                        LOGGER.error("RPCAP packet read error: %s", e, exc_info=True)
                        self.error_occurred.emit(f"Read error: {str(e)}")
                    break
            
            self.status_message.emit(f"RPCAP capture finished: {self.packet_counter} packets")
            
        except Exception as e:
            LOGGER.error("RPCAP capture thread error: %s", e, exc_info=True)
            self.error_occurred.emit(f"Thread error: {str(e)}")
        finally:
            self.running = False
    
    def stop(self):
        """Stop the capture thread"""
        self.running = False
        if self.isRunning():
            self.wait(3000)  # Wait up to 3 seconds

class RpcapClient:
    """Pure-Python RPCAP client matching Wireshark's RPCAP protocol"""
    
    # RPCAP Protocol Constants (matching Wireshark/Npcap)
    RPCAP_VERSION = 0
    RPCAP_MSG_FINDALLIF_REQ = 1
    RPCAP_MSG_OPEN_REQ = 2
    RPCAP_MSG_STARTCAP_REQ = 3
    RPCAP_MSG_UPDATEFILTER_REQ = 4
    RPCAP_MSG_CLOSE = 5
    RPCAP_MSG_PACKET = 6
    RPCAP_MSG_AUTH_REQ = 7
    RPCAP_MSG_STATS_REQ = 8
    RPCAP_MSG_ENDCAP_REQ = 9
    RPCAP_MSG_SETSAMPLING_REQ = 10
    
    RPCAP_MSG_FINDALLIF_REPLY = 128 + 1
    RPCAP_MSG_OPEN_REPLY = 128 + 2
    RPCAP_MSG_STARTCAP_REPLY = 128 + 3
    RPCAP_MSG_UPDATEFILTER_REPLY = 128 + 4
    RPCAP_MSG_AUTH_REPLY = 128 + 7
    RPCAP_MSG_STATS_REPLY = 128 + 8
    RPCAP_MSG_ENDCAP_REPLY = 128 + 9
    RPCAP_MSG_SETSAMPLING_REPLY = 128 + 10
    RPCAP_MSG_ERROR = 128 + 11
    
    def __init__(self, host, port=2002, password=None):
        self.host = host
        self.port = port
        self.password = password
        self.socket = None
        self.connected = False
        self.packet_counter = 0
        self.capture_thread = None  # Will be RPCAPCaptureThread (QThread)
        self.running = False
        self.packet_callback = None
        
    def connect(self):
        """Connect to rpcapd server (Wireshark-compatible)"""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.settimeout(10)  # 10 second timeout for connection
            LOGGER.info(f"Attempting RPCAP connection to {self.host}:{self.port}")
            self.socket.connect((self.host, self.port))
            LOGGER.info(f"TCP connection established to {self.host}:{self.port}")
            
            # FIXED: Server sends version first, client just receives it (no client response needed)
            # Server sends 4 bytes: version (1 byte) + reserved (3 bytes)
            response = self.socket.recv(4)
            if len(response) < 4:
                LOGGER.error("RPCAP server version response too short")
                self.socket.close()
                self.socket = None
                return False
            
            # Server sends 4 bytes: version (1 byte) + reserved (3 bytes)
            server_version = response[0]  # Just first byte is the version
            LOGGER.info(f"RPCAP server version: {server_version} (host={self.host}, port={self.port})")
            
            # No version sent by client - proceed directly to commands
            
            # Handle authentication if password provided
            if self.password:
                # FIXED: Use correct header format !BBHI
                auth_payload = self.password.encode('utf-8')
                auth_header = struct.pack('!BBHI', 0, self.RPCAP_MSG_AUTH_REQ, 0, len(auth_payload))
                auth_msg = auth_header + auth_payload
                self.socket.sendall(auth_msg)
                
                # Read AUTH response (8 bytes: RPCAP header)
                auth_response = self.socket.recv(8)
                if len(auth_response) < 8:
                    LOGGER.error("RPCAP authentication response too short")
                    self.socket.close()
                    self.socket = None
                    return False
                
                auth_ver, auth_type, auth_value, auth_plen = struct.unpack('!BBHI', auth_response)
                if auth_type != self.RPCAP_MSG_AUTH_REPLY or auth_value != 0:
                    LOGGER.error(f"RPCAP authentication failed (response type={auth_type}, expected={self.RPCAP_MSG_AUTH_REPLY}, value={auth_value})")
                    self.socket.close()
                    self.socket = None
                    return False
                LOGGER.info("RPCAP authentication successful")
            
            self.connected = True
            LOGGER.info(f"RPCAP connection established to {self.host}:{self.port}")
            return True
        except Exception as e:
            LOGGER.error(f"RPCAP connection failed to {self.host}:{self.port}: {e}", exc_info=True)
            if self.socket:
                try:
                    self.socket.close()
                except:
                    pass
                self.socket = None  # FIXED: Set to None after close
            return False
    
    def list_interfaces(self):
        """List available interfaces using RPCAP findalldevs (Wireshark-compatible)"""
        if not self.connected:
            return []
            
        try:
            # Send RPCAP findalldevs request
            # RPCAP header format: !BBHI (version:1, type:1, value:2, plen:4)
            # FINDALLIF_REQ: version=0, type=FINDALLIF_REQ, value=0, plen=0
            request = struct.pack('!BBHI', 0, self.RPCAP_MSG_FINDALLIF_REQ, 0, 0)
            self.socket.send(request)
            
            # Read response header (8 bytes: RPCAP header)
            header = self.socket.recv(8)
            if len(header) < 8:
                return []
            
            ver, msg_type, value, plen = struct.unpack('!BBHI', header)
            if msg_type != self.RPCAP_MSG_FINDALLIF_REPLY:
                LOGGER.warning(f"Unexpected RPCAP response type: {msg_type} (expected {self.RPCAP_MSG_FINDALLIF_REPLY})")
                return []
            
            # Read number of interfaces (4 bytes) from payload
            nif_data = self.socket.recv(4)
            if len(nif_data) < 4:
                return []
            nif = struct.unpack('!I', nif_data)[0]
            
            interfaces = []
            
            # Read interface list
            for _ in range(nif):
                # Read interface name length
                name_len_data = self.socket.recv(4)
                if len(name_len_data) < 4:
                    break
                name_len = struct.unpack('!I', name_len_data)[0]
                
                # Read interface name
                name_data = self.socket.recv(name_len)
                if len(name_data) < name_len:
                    break
                name = name_data.decode('utf-8', errors='ignore').rstrip('\x00')
                
                # Read interface description length
                desc_len_data = self.socket.recv(4)
                if len(desc_len_data) < 4:
                    break
                desc_len = struct.unpack('!I', desc_len_data)[0]
                
                # Skip description
                if desc_len > 0:
                    self.socket.recv(desc_len)
                
                # Read flags (4 bytes) and skip
                self.socket.recv(4)
                
                # Read address count (4 bytes)
                addr_count_data = self.socket.recv(4)
                if len(addr_count_data) < 4:
                    break
                addr_count = struct.unpack('!I', addr_count_data)[0]
                
                # Skip addresses
                for _ in range(addr_count):
                    # Skip address family (2 bytes) + padding (2 bytes)
                    self.socket.recv(4)
                    # Skip address length (4 bytes)
                    addr_len_data = self.socket.recv(4)
                    if len(addr_len_data) < 4:
                        break
                    addr_len = struct.unpack('!I', addr_len_data)[0]
                    # Skip address data
                    self.socket.recv(addr_len)
                    # Skip netmask length (4 bytes)
                    netmask_len_data = self.socket.recv(4)
                    if len(netmask_len_data) < 4:
                        break
                    netmask_len = struct.unpack('!I', netmask_len_data)[0]
                    # Skip netmask data
                    self.socket.recv(netmask_len)
                    # Skip broadcast address length (4 bytes)
                    broadcast_len_data = self.socket.recv(4)
                    if len(broadcast_len_data) < 4:
                        break
                    broadcast_len = struct.unpack('!I', broadcast_len_data)[0]
                    # Skip broadcast address data
                    self.socket.recv(broadcast_len)
                    # Skip destination address length (4 bytes)
                    dest_len_data = self.socket.recv(4)
                    if len(dest_len_data) < 4:
                        break
                    dest_len = struct.unpack('!I', dest_len_data)[0]
                    # Skip destination address data
                    self.socket.recv(dest_len)
                
                if name and name not in interfaces:
                    interfaces.append(name)
                    
            LOGGER.info(f"RPCAP found {len(interfaces)} interfaces: {interfaces}")
            return interfaces
            
        except Exception as e:
            LOGGER.error(f"RPCAP list interfaces failed: {e}", exc_info=True)
            return []
    
    def start_capture(self, interface, packet_callback):
        """Start capture on specified interface (Wireshark-compatible RPCAP protocol)"""
        if not self.connected:
            return False
        
        self.packet_callback = packet_callback
        self.running = True
            
        try:
            # === OPEN REQUEST (FIXED: Correct RPCAP protocol format) ===
            # RPCAP header format: !BBHI (version:1, type:1, value:2, plen:4)
            # Payload: null-terminated interface name
            iface_bytes = interface.encode('utf-8') + b'\0'  # MUST be null-terminated
            open_header = struct.pack("!BBHI", 0, self.RPCAP_MSG_OPEN_REQ, 0, len(iface_bytes))
            open_req = open_header + iface_bytes
            self.socket.sendall(open_req)  # Use sendall to ensure all data is sent
            
            # Read OPEN response (8 bytes: RPCAP header)
            reply = self.socket.recv(8)
            if len(reply) < 8:
                LOGGER.error("RPCAP: OPEN reply too short (expected 8 bytes)")
                self.running = False
                return False
            
            ver, msg_type, value, plen = struct.unpack("!BBHI", reply)
            
            # Check for error message
            if msg_type == self.RPCAP_MSG_ERROR:
                if plen > 0:
                    err_data = self.socket.recv(plen)
                    err_msg = err_data.decode('utf-8', errors='ignore')
                    LOGGER.error(f"RPCAP OPEN error: {err_msg}")
                else:
                    LOGGER.error("RPCAP OPEN empty error (protocol mismatch?)")
                self.running = False
                return False
            
            # Success check - OPEN_REPLY must have plen=0 (no payload on success)
            if msg_type != self.RPCAP_MSG_OPEN_REPLY or plen != 0:
                LOGGER.error(f"RPCAP: OPEN failed (type={msg_type}, expected={self.RPCAP_MSG_OPEN_REPLY}, plen={plen}, expected_plen=0, interface={interface})")
                # Discard any payload if present
                if plen > 0:
                    self._discard_payload(plen)
                self.running = False
                return False
            LOGGER.info(f"RPCAP: Interface '{interface}' opened successfully")
            
            # === STARTCAP REQUEST (FIXED: Correct RPCAP protocol format) ===
            # RPCAP header format: !BBHI (version:1, type:1, value:2, plen:4)
            # Payload: snaplen (uint32, 4 bytes) + read_timeout (uint32, 4 bytes) + flags (uint32, 4 bytes) + packet_count (uint32, 4 bytes)
            # Total payload: 16 bytes
            snaplen = 65535
            read_timeout = 1000  # milliseconds
            flags = 1  # 1 = promiscuous mode (critical for packet capture!)
            packet_count = 0  # 0 = infinite packets
            
            # FIXED: Use !IIII format (uint32, uint32, uint32, uint32) = 16 bytes total
            payload = struct.pack("!IIII", snaplen, read_timeout, flags, packet_count)  # 16 bytes total
            start_header = struct.pack("!BBHI", 0, self.RPCAP_MSG_STARTCAP_REQ, 0, len(payload))
            start_req = start_header + payload
            self.socket.sendall(start_req)  # Use sendall to ensure all data is sent
            
            LOGGER.debug(f"RPCAP: Sent STARTCAP_REQ (snaplen={snaplen}, timeout={read_timeout}ms, flags={flags}, packet_count={packet_count}, payload_len={len(payload)})")
            
            # Read STARTCAP response (8 bytes: RPCAP header)
            reply = self.socket.recv(8)
            if len(reply) < 8:
                LOGGER.error("RPCAP: STARTCAP reply too short (expected 8 bytes)")
                self.running = False
                return False
            
            ver, msg_type, value, plen = struct.unpack("!BBHI", reply)
            
            # FIXED: Proper error handling - check for ERROR message type
            if msg_type == self.RPCAP_MSG_ERROR:
                if plen > 0:
                    # Read error message payload
                    err_data = self.socket.recv(plen)
                    err_msg = err_data.decode('utf-8', errors='ignore')
                    LOGGER.error(f"RPCAP STARTCAP error: {err_msg}")
                else:
                    LOGGER.error("RPCAP STARTCAP empty error (protocol mismatch?)")
                self.running = False
                return False
            
            # FIXED: Success check - STARTCAP_REPLY must have value=0 (success indicator)
            if msg_type != self.RPCAP_MSG_STARTCAP_REPLY or value != 0:
                LOGGER.error(f"RPCAP: STARTCAP failed (type={msg_type}, expected={self.RPCAP_MSG_STARTCAP_REPLY}, value={value}, expected_value=0, interface={interface})")
                # Discard any payload if present
                if plen > 0:
                    self._discard_payload(plen)
                self.running = False
                return False
            
            # Read STARTCAP_REPLY payload if present (usually 4 bytes containing buffer size)
            if plen > 0:
                buffersize_data = self.socket.recv(plen)
                if len(buffersize_data) == 4:
                    buffer_size = struct.unpack('!I', buffersize_data)[0]
                    LOGGER.debug(f"RPCAP: STARTCAP_REPLY buffer size: {buffer_size}")
                else:
                    LOGGER.warning(f"RPCAP: STARTCAP_REPLY payload unexpected length: {len(buffersize_data)}")
                
            LOGGER.info(f"RPCAP capture started successfully on interface '{interface}' (snaplen={snaplen}, timeout={read_timeout}ms, flags={flags})")
            
            # Start capture loop in QThread for thread safety
            # FIXED: Don't pass callback to constructor if connecting signal afterward
            self.capture_thread = RPCAPCaptureThread(self.socket, self, None)
            self.capture_thread.packet_received.connect(packet_callback)
            self.capture_thread.error_occurred.connect(lambda msg: LOGGER.error(f"RPCAP capture thread error: {msg}"))
            self.capture_thread.status_message.connect(lambda msg: LOGGER.info(f"RPCAP: {msg}"))
            self.capture_thread.start()
            
            return True
            
        except Exception as e:
            LOGGER.error(f"RPCAP start_capture exception on interface '{interface}': {e}", exc_info=True)
            self.running = False
            return False
    
    def _capture_loop(self):
        """Capture loop - read packets from RPCAP stream (runs in thread)"""
        try:
            self.packet_counter = 0
            while self.running and self.connected:
                # Read message header (8 bytes: RPCAP protocol header)
                # Format: !BBHI (version:1, type:1, value:2, plen:4)
                header = self.socket.recv(8)
                if len(header) < 8:
                    break
                
                ver, msg_type, value, msg_length = struct.unpack('!BBHI', header)
                
                if msg_type == self.RPCAP_MSG_PACKET:
                    # Read packet header (20 bytes: RPCAP packet header format)
                    # Format: timestamp_sec (4) + timestamp_usec (4) + caplen (4) + len (4) + npkt (4)
                    pkt_header = self.socket.recv(20)
                    if len(pkt_header) < 20:
                        break
                    
                    # Parse packet header
                    timestamp_sec = struct.unpack('!I', pkt_header[0:4])[0]
                    timestamp_usec = struct.unpack('!I', pkt_header[4:8])[0]
                    caplen = struct.unpack('!I', pkt_header[8:12])[0]
                    len_pkt = struct.unpack('!I', pkt_header[12:16])[0]
                    npkt = struct.unpack('!I', pkt_header[16:20])[0]  # Packet number
                    
                    # Read packet data
                    packet_data = b''
                    while len(packet_data) < caplen:
                        chunk = self.socket.recv(min(caplen - len(packet_data), 4096))
                        if not chunk:
                            break
                        packet_data += chunk
                    
                    if len(packet_data) == caplen:
                        timestamp = timestamp_sec + timestamp_usec / 1000000.0
                        packet_info = None
                        try:
                            scapy_packet = Ether(packet_data)
                            packet_info = parse_packet_from_scapy(
                                scapy_packet,
                                timestamp=timestamp,
                                sip_ports=DEFAULT_SIP_PORTS,
                                rtp_port_range=DEFAULT_RTP_PORT_RANGE
                            )
                        except Exception as parse_error:
                            LOGGER.debug("RPCAP packet parse error: %s", parse_error, exc_info=True)

                        if not packet_info:
                            packet_info = {
                                'timestamp': _format_packet_timestamp(timestamp),
                                'src': '',
                                'dst': '',
                                'sport': 0,
                                'dport': 0,
                                'protocol': 'Unknown',
                                'length': len_pkt,
                                'qos_dscp': 0,
                                'qos_name': 'BE',
                                'flags': '',
                                'info': 'RPCAP packet'
                            }

                        self.packet_counter += 1
                        packet_info['no'] = self.packet_counter
                        packet_info['frame_number'] = self.packet_counter
                        packet_info['length'] = len_pkt
                        packet_info['raw_packet'] = packet_data

                        if self.packet_callback:
                            self.packet_callback(packet_info)
                elif msg_type == self.RPCAP_MSG_ERROR:
                    # Error message payload (plen bytes)
                    if msg_length > 0:
                        error_data = self.socket.recv(msg_length)
                        error_msg = error_data.decode('utf-8', errors='ignore')
                        LOGGER.error(f"RPCAP server error message: {error_msg}")
                    else:
                        LOGGER.error("RPCAP server sent error message with no details")
                    break
                else:
                    # Unknown message type, skip payload
                    if msg_length > 0:
                        self.socket.recv(msg_length)
                        
        except Exception as e:
            if self.running:
                LOGGER.error(f"RPCAP capture loop error: {e}", exc_info=True)
    
    def _discard_payload(self, plen):
        """Helper to discard payload bytes (e.g., when handling errors)"""
        if plen > 0:
            junk = b''
            while len(junk) < plen:
                chunk = self.socket.recv(min(plen - len(junk), 1024))
                if not chunk:
                    break
                junk += chunk
    
    def stop_capture(self):
        """Stop capture"""
        self.running = False
        # FIXED: Only send ENDCAP if capture was actually started
        if self.socket and self.connected and hasattr(self, 'capture_thread') and self.capture_thread:
            try:
                # Send end capture request (RPCAP header format: !BBHI)
                end_req = struct.pack('!BBHI', 0, self.RPCAP_MSG_ENDCAP_REQ, 0, 0)
                self.socket.sendall(end_req)  # Use sendall to ensure all data is sent
            except Exception as e:
                LOGGER.debug("Error sending RPCAP end request: %s", e)
        
        if self.capture_thread:
            if isinstance(self.capture_thread, QThread):
                # QThread - use Qt methods
                if hasattr(self.capture_thread, 'stop'):
                    self.capture_thread.stop()
                if self.capture_thread.isRunning():
                    self.capture_thread.wait(3000)
            else:
                # Regular thread - use threading methods
                self.capture_thread.join(timeout=2)
            self.capture_thread = None
    
    def disconnect(self):
        """Disconnect from server"""
        self.stop_capture()
        self.connected = False
        if self.socket:
            try:
                self.socket.close()
            except:
                pass
            self.socket = None

def setup_logging():
    """Setup logging with daily file rotation and comprehensive error logging"""
    # Create logger
    logger = logging.getLogger('NetHawk')
    logger.setLevel(logging.DEBUG)  # Capture all levels
    
    # Remove any existing handlers to avoid duplicates
    logger.handlers.clear()

        os.makedirs(PROGRAM_DATA_DIR, exist_ok=True)
    except OSError:
        pass

        '%(asctime)s [%(levelname)-8s] [%(filename)s:%(lineno)d] %(funcName)s() - %(message)s',

    )

    # Creates files like: nethawk_20240115.log, nethawk_20240116.log, etc.
    # Rotates at midnight, keeps 30 days of logs
    
    # Create a custom namer function to format log files as: nethawk_YYYYMMDD.log
    def namer(name):
        """Custom namer to create log files with format: nethawk_YYYYMMDD.log"""
        # TimedRotatingFileHandler passes the rotated filename
        # Default format is: base.YYYYMMDD -> we want: base_YYYYMMDD.log
        # Handle both cases: base.YYYYMMDD and base_YYYYMMDD.log
        if name.endswith('.log'):
            # Already has .log extension
            if '.' in name[:-4]:  # Check before .log
                # Replace first dot with underscore: nethawk.20240115.log -> nethawk_20240115.log
                base_part = name[:-4]  # Everything before .log
                if '.' in base_part:
                    base, date = base_part.split('.', 1)
                    return f"{base}_{date}.log"
        else:
            # No .log extension yet, add it
            if '.' in name:
                base, date = name.split('.', 1)
                return f"{base}_{date}.log"
            else:
                return f"{name}.log"
        return name
    
    file_handler = TimedRotatingFileHandler(
        filename=LOG_FILE_BASE,  # Base name: nethawk (without extension)
        when='midnight',
        interval=1,
        backupCount=30,  # Keep 30 days of logs
        encoding='utf-8',
        delay=False
    )
    # Override the namer to use underscore format: nethawk_YYYYMMDD.log
    file_handler.namer = namer
    # Set suffix to date format (used for rotation)
    file_handler.suffix = '%Y%m%d'
    # Override the baseFilename to include .log extension for current file
    # This ensures the initial file is created as nethawk_YYYYMMDD.log

    current_date = time.strftime('%Y%m%d')
    file_handler.baseFilename = f"{LOG_FILE_BASE}_{current_date}.log"

    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)
    
    # Console handler for critical errors (optional, but helpful)
    console_handler = logging.StreamHandler(sys.stderr)
    console_handler.setLevel(logging.WARNING)  # Only warnings and errors to console
    console_formatter = logging.Formatter(
        '%(asctime)s [%(levelname)s] %(message)s',
        datefmt='%H:%M:%S'
    )
    console_handler.setFormatter(console_formatter)
    logger.addHandler(console_handler)
    
    # Log startup message
    logger.info("=" * 80)
    logger.info("NetHawk Application Started")
    logger.info(f"Log file: {LOG_FILE}")
    logger.info(f"Program data directory: {PROGRAM_DATA_DIR}")
    logger.info(f"Python version: {sys.version}")
    logger.info("=" * 80)
    
    return logger

def _format_packet_timestamp(ts=None):
    if ts is None:
        return datetime.now().strftime('%H:%M:%S.%f')[:-3]
    if isinstance(ts, (int, float)):
        return datetime.fromtimestamp(ts).strftime('%H:%M:%S.%f')[:-3]
    return str(ts)


def detect_application_protocol(packet, sport, dport, transport, sip_ports=None, rtp_port_range=None):
    sip_ports = sip_ports or DEFAULT_SIP_PORTS
    rtp_port_range = rtp_port_range or DEFAULT_RTP_PORT_RANGE

    port_map = {
        21: 'FTP', 22: 'SSH', 25: 'SMTP', 53: 'DNS',
        80: 'HTTP', 443: 'HTTPS', 5060: 'SIP', 5061: 'SIP'
    }

    if sport in port_map:
        return port_map[sport]
    if dport in port_map:
        return port_map[dport]

    if sport in sip_ports or dport in sip_ports:
        if packet.haslayer(Raw):
            try:
                payload = bytes(packet[Raw].load)
                payload_str = payload.decode('utf-8', errors='ignore')[:200]
                sip_markers = ['SIP/2.0', 'INVITE', 'REGISTER', 'ACK', 'BYE', 'CANCEL', 'OPTIONS']
                if any(marker in payload_str for marker in sip_markers):
                    return 'SIP'
            except Exception:
                pass

    if transport == 'UDP' and rtp_port_range:
        low, high = rtp_port_range
        if (low <= sport <= high) or (low <= dport <= high):
            if packet.haslayer(Raw):
                payload = bytes(packet[Raw].load)
                if len(payload) >= 12:
                    version = (payload[0] >> 6) & 0x03
                    payload_type = payload[1] & 0x7F
                    if version == 2 and payload_type <= 127:
                        return 'RTP'

    if packet.haslayer(Raw):
        try:
            payload = bytes(packet[Raw].load)
            payload_str = payload.decode('utf-8', errors='ignore')[:50]
            if any(marker in payload_str for marker in ['GET ', 'POST ', 'HTTP/']):
                return 'HTTP'
        except Exception:
            pass

    return transport


def parse_packet_from_scapy(packet, timestamp=None, sip_ports=None, rtp_port_range=None):
    try:
        packet_info = {
            'timestamp': _format_packet_timestamp(timestamp),
            'src': 'Unknown',
            'dst': 'Unknown',
            'sport': 0,
            'dport': 0,
            'protocol': 'Other',
            'length': len(packet),
            'qos_dscp': 0,
            'qos_name': 'BE',
            'flags': '',
            'info': ''
        }

        if packet.haslayer(ARP):
            arp = packet[ARP]
            packet_info.update({
                'protocol': 'ARP',
                'src': arp.psrc,
                'dst': arp.pdst,
                'info': f"ARP {arp.psrc} → {arp.pdst}"
            })
            return packet_info

        if packet.haslayer(IP):
            ip_layer = packet[IP]
            packet_info['src'] = ip_layer.src
            packet_info['dst'] = ip_layer.dst
            packet_info['qos_dscp'] = (ip_layer.tos >> 2) & 0x3F
        elif packet.haslayer(IPv6):
            ipv6_layer = packet[IPv6]
            packet_info['src'] = ipv6_layer.src
            packet_info['dst'] = ipv6_layer.dst
            traffic_class = getattr(ipv6_layer, 'tc', 0)
            packet_info['qos_dscp'] = (traffic_class >> 2) & 0x3F
        else:
            return None

        dscp_entry = DSCP_CLASSES.get(packet_info['qos_dscp'])
        packet_info['qos_name'] = dscp_entry[1] if dscp_entry else 'BE'

        if packet.haslayer(TCP):
            tcp = packet[TCP]
            packet_info['sport'] = tcp.sport
            packet_info['dport'] = tcp.dport
            app_proto = detect_application_protocol(packet, tcp.sport, tcp.dport, 'TCP', sip_ports, rtp_port_range)
            packet_info['protocol'] = app_proto
            flags = []
            try:
                if tcp.flags.F:
                    flags.append('FIN')
                if tcp.flags.S:
                    flags.append('SYN')
                if tcp.flags.R:
                    flags.append('RST')
                if tcp.flags.P:
                    flags.append('PSH')
                if tcp.flags.A:
                    flags.append('ACK')
            except Exception:
                pass
            packet_info['flags'] = ','.join(flags)
            packet_info['info'] = (
                f"{packet_info['protocol']} [{packet_info['flags']}]"
                if packet_info['flags'] else packet_info['protocol']
            )
        elif packet.haslayer(UDP):
            udp = packet[UDP]
            packet_info['sport'] = udp.sport
            packet_info['dport'] = udp.dport
            app_proto = detect_application_protocol(packet, udp.sport, udp.dport, 'UDP', sip_ports, rtp_port_range)
            packet_info['protocol'] = app_proto
            packet_info['info'] = f"{packet_info['protocol']} {udp.sport}→{udp.dport}"
        elif packet.haslayer(ICMP):
            packet_info['protocol'] = 'ICMP'
            packet_info['info'] = 'ICMP'
        else:
            summary = packet.summary().split()
            if summary:
                packet_info['protocol'] = summary[0]

        return packet_info
    except Exception:
        return None


class FilterParser:
    """Wireshark-style filter parser"""
    
    @staticmethod
    def parse_and_evaluate(filter_text, packet_info):
        """Parse and evaluate filter expression"""
        if not filter_text or not filter_text.strip():
            return True
        
        filter_text = filter_text.strip().lower()
        
        # Handle logical operators
        if ' or ' in filter_text:
            parts = filter_text.split(' or ')
            return any(FilterParser.parse_and_evaluate(p, packet_info) for p in parts)
        
        if ' and ' in filter_text:
            parts = filter_text.split(' and ')
            return all(FilterParser.parse_and_evaluate(p, packet_info) for p in parts)
        
        if filter_text.startswith('not '):
            return not FilterParser.parse_and_evaluate(filter_text[4:], packet_info)
        
        # Remove parentheses for simple parsing
        filter_text = filter_text.replace('(', '').replace(')', '')
        
        # Parse comparison expressions
        operators = ['==', '!=', 'contains', '>=', '<=', '>', '<']
        for op in operators:
            if op in filter_text:
                parts = filter_text.split(op, 1)
                if len(parts) == 2:
                    field = parts[0].strip()
                    value = parts[1].strip().strip('"').strip("'")
                    return FilterParser.evaluate_comparison(field, op, value, packet_info)
        
        # Fallback: simple text search
        return FilterParser.simple_search(filter_text, packet_info)
    
    @staticmethod
    def evaluate_comparison(field, operator, value, packet_info):
        """Evaluate field comparison"""
        # IP address fields
        if field in ['ip.src', 'src', 'source']:
            packet_val = packet_info.get('src', '').lower()
        elif field in ['ip.dst', 'dst', 'destination']:
            packet_val = packet_info.get('dst', '').lower()
        elif field in ['ip.addr', 'ip', 'addr']:
            src = packet_info.get('src', '').lower()
            dst = packet_info.get('dst', '').lower()
            if operator == '==':
                return value in [src, dst]
            elif operator == '!=':
                return value not in [src, dst]
            elif operator == 'contains':
                return value in src or value in dst
            return False
        
        # Port fields
        elif field in ['tcp.port', 'udp.port', 'port']:
            sport = str(packet_info.get('sport', 0))
            dport = str(packet_info.get('dport', 0))
            if operator == '==':
                return value in [sport, dport]
            elif operator == '!=':
                return value not in [sport, dport]
            try:
                val_int = int(value)
                return val_int in [int(sport), int(dport)]
            except:
                return False
        
        elif field in ['tcp.srcport', 'udp.srcport', 'sport', 'src.port']:
            packet_val = str(packet_info.get('sport', 0))
        elif field in ['tcp.dstport', 'udp.dstport', 'dport', 'dst.port']:
            packet_val = str(packet_info.get('dport', 0))
        
        # Protocol field
        elif field in ['protocol', 'proto']:
            packet_val = packet_info.get('protocol', '').lower()
        
        # Length field
        elif field in ['length', 'len', 'frame.len']:
            try:
                packet_val = packet_info.get('length', 0)
                value_num = int(value)
                if operator == '==':
                    return packet_val == value_num
                elif operator == '!=':
                    return packet_val != value_num
                elif operator == '>':
                    return packet_val > value_num
                elif operator == '<':
                    return packet_val < value_num
                elif operator == '>=':
                    return packet_val >= value_num
                elif operator == '<=':
                    return packet_val <= value_num
            except:
                return False
            return False
        
        # Info field
        elif field in ['info', 'description']:
            packet_val = packet_info.get('info', '').lower()
        
        else:
            return False
        
        # Perform comparison
        if operator == '==':
            return packet_val == value
        elif operator == '!=':
            return packet_val != value
        elif operator == 'contains':
            return value in packet_val
        elif operator in ['>', '<', '>=', '<=']:
            try:
                return eval(f"{packet_val} {operator} {value}")
            except:
                return False
        
        return False
    
    @staticmethod
    def simple_search(text, packet_info):
        """Simple text search fallback"""
        text = text.lower()
        
        if packet_info.get('protocol', '').lower() == text:
            return True
        if text in packet_info.get('src', '').lower():
            return True
        if text in packet_info.get('dst', '').lower():
            return True
        if text in packet_info.get('info', '').lower():
            return True
        
        return False

def ulaw2lin(data):
    """Convert μ-law to linear PCM"""
    ULAW_BIAS = 0x84
    output = bytearray()
    for byte_val in data:
        byte_val = ~byte_val & 0xFF
        sign = byte_val & 0x80
        exponent = (byte_val >> 4) & 0x07
        mantissa = byte_val & 0x0F
        linear = ((mantissa << 3) + ULAW_BIAS) << exponent
        linear -= ULAW_BIAS
        if sign:
            linear = -linear
        linear = max(-32768, min(32767, linear))
        output.extend(struct.pack('<h', linear))
    return bytes(output)

def alaw2lin(data):
    """Convert A-law to linear PCM"""
    output = bytearray()
    for byte_val in data:
        byte_val ^= 0x55
        sign = byte_val & 0x80
        exponent = (byte_val >> 4) & 0x07
        mantissa = byte_val & 0x0F
        if exponent == 0:
            linear = (mantissa << 4) + 8
        else:
            linear = ((mantissa << 4) + 0x108) << (exponent - 1)
        if not sign:
            linear = -linear
        linear = max(-32768, min(32767, linear))
        output.extend(struct.pack('<h', linear))
    return bytes(output)

class PacketDatabase:
    """SQLite database for packet storage"""
    def __init__(self, db_file=DB_FILE):
        self.db_file = db_file
        self.init_database()
    
    def init_database(self):
        """Initialize database"""
        conn = sqlite3.connect(self.db_file)
        cursor = conn.cursor()

class RTPStream:
    """RTP stream handler"""
    def __init__(self, ssrc, payload_type=0):
        self.ssrc = ssrc
        self.payload_type = payload_type
        self.packets = []
        self.codec = CODEC_INFO.get(payload_type, {'name': 'Unknown', 'rate': 8000})
        self.sample_rate = self.codec['rate']
        self.start_time = None
        self.end_time = None
        self.packet_count = 0
        self.lost_packets = 0
        self.jitter_samples = []
        self.max_jitter = 0
        self.avg_jitter = 0
    
    def add_packet(self, rtp_packet, timestamp):
        if self.start_time is None:
            self.start_time = timestamp
        self.end_time = timestamp
        self.packets.append({
            'seq': rtp_packet.sequence,
            'timestamp': rtp_packet.timestamp,
            'payload': bytes(rtp_packet.payload),
            'received_at': timestamp
        })
        self.packet_count += 1
        if len(self.packets) > 1:
            self.calculate_jitter()
    
    def calculate_jitter(self):
        if len(self.packets) < 2:
            return
        p1 = self.packets[-2]
        p2 = self.packets[-1]
        arrival_diff = (p2['received_at'] - p1['received_at']) * 1000
        timestamp_diff = (p2['timestamp'] - p1['timestamp']) / (self.sample_rate / 1000.0)
        jitter = abs(arrival_diff - timestamp_diff)
        self.jitter_samples.append(jitter)
        if jitter > self.max_jitter:
            self.max_jitter = jitter
        if self.jitter_samples:
            self.avg_jitter = sum(self.jitter_samples) / len(self.jitter_samples)
    
    def detect_packet_loss(self):
        if len(self.packets) < 2:
            return
        sequences = [p['seq'] for p in self.packets]
        sequences.sort()
        expected_count = sequences[-1] - sequences[0] + 1
        self.lost_packets = expected_count - len(sequences)
    
    def calculate_mos(self):
        self.detect_packet_loss()
        if self.packet_count == 0:
            return 0.0
        loss_rate = (self.lost_packets / (self.packet_count + self.lost_packets)) * 100
        mos = 4.5
        if loss_rate > 0:
            mos -= min(loss_rate * 0.1, 3.0)
        if self.avg_jitter > 20:
            mos -= min((self.avg_jitter - 20) * 0.02, 1.5)
        return max(1.0, min(4.5, mos))
    
    def export_to_wav(self, filename):
        try:
            sorted_packets = sorted(self.packets, key=lambda x: x['seq'])
            audio_data = b''.join([p['payload'] for p in sorted_packets])
            if not audio_data:
                return False
            if self.payload_type == 0:
                pcm_data = ulaw2lin(audio_data)
            elif self.payload_type == 8:
                pcm_data = alaw2lin(audio_data)
            else:
                pcm_data = audio_data
            with wave.open(filename, 'wb') as wav_file:
                wav_file.setnchannels(1)
                wav_file.setsampwidth(2)
                wav_file.setframerate(self.sample_rate)
                wav_file.writeframes(pcm_data)
            return True
        except Exception as e:
            print(f"WAV export error: {e}")
            return False

class TSharkPacketReaderThread(QThread):
    """QThread to safely read packets from TShark output (Wireshark-compatible)"""
    packet_received = pyqtSignal(dict)
    error_occurred = pyqtSignal(str)
    status_message = pyqtSignal(str)
    
    def __init__(self, process, callback=None):
        super().__init__()
        self.process = process
        self.callback = callback
        self.running = False
        self.packet_counter = 0
    
    def run(self):
        """Read TShark field output and parse packets - Wireshark-compatible"""
        self.running = True
        try:
            # Read stderr in a separate thread to avoid blocking
            def stderr_reader():
                try:
                    for line in iter(self.process.stderr.readline, ''):
                        if line.strip() and self.running:
                            self.error_occurred.emit(f"TShark stderr: {line.strip()}")
                except:
                    pass
            
            stderr_thread = threading.Thread(target=stderr_reader, daemon=True)
            stderr_thread.start()
            
            self.status_message.emit("TShark reader started...")
            
            line_count = 0
            for line in self.process.stdout:
                if not self.running:
                    self.status_message.emit("TShark reader stopped")
                    break
                    
                line_count += 1
                
                try:
                    line = line.strip()
                    if not line:
                        continue
                        
                    fields = line.split('\t')
                    if len(fields) >= 8:
                        packet_info = self.parse_tshark_fields(fields)
                        if packet_info:
                            self.packet_counter += 1
                            # Emit packet safely via Qt signal (thread-safe)
                            self.packet_received.emit(packet_info)
                            
                            if self.packet_counter % 100 == 0:
                                self.status_message.emit(f"Processed {self.packet_counter} packets")
                    else:
                        if line_count <= 10:  # Log first 10 invalid lines
                            self.error_occurred.emit(f"Invalid line format (fields={len(fields)})")
                            
                except Exception as e:
                    LOGGER.debug("TShark parsing error: %s", e, exc_info=True)
                    continue
            
            self.status_message.emit(f"TShark finished: {self.packet_counter} packets from {line_count} lines")
            
        except Exception as e:
            LOGGER.error("TShark reader thread error: %s", e, exc_info=True)
            self.error_occurred.emit(f"Thread error: {str(e)}")
        finally:
            self.running = False
    
    def parse_tshark_fields(self, fields):
        """Parse tshark output fields"""
        try:
            # Extract ports
            src_port = fields[5] if len(fields) > 5 else ''
            dst_port = fields[6] if len(fields) > 6 else ''
            
            # Determine protocol name
            protocol_num = fields[4] if len(fields) > 4 else '0'
            protocol_name = 'Unknown'
            if protocol_num == '6':
                protocol_name = 'TCP'
            elif protocol_num == '17':
                protocol_name = 'UDP'
            elif protocol_num == '1':
                protocol_name = 'ICMP'
            elif protocol_num == '2':
                protocol_name = 'IGMP'
                
            return {
                'timestamp': fields[1] if len(fields) > 1 else str(datetime.now()),
                'src': fields[2] if len(fields) > 2 else '',
                'dst': fields[3] if len(fields) > 3 else '',
                'protocol': protocol_name,
                'length': int(fields[7]) if len(fields) > 7 and fields[7].isdigit() else 0,
                'info': f"{protocol_name} {src_port} -> {dst_port}",
                'frame_number': int(fields[0]) if len(fields) > 0 and fields[0].isdigit() else 0,
                'raw_packet': b'',
                'sport': int(src_port) if src_port.isdigit() else 0,
                'dport': int(dst_port) if dst_port.isdigit() else 0
            }
        except Exception as e:
            LOGGER.debug("TShark field parse error: %s", e)
            return None
    
    def stop(self):
        """Stop the reader thread"""
        self.running = False
        if self.isRunning():
            self.wait(3000)  # Wait up to 3 seconds

class SSHPacketReaderThread(QThread):
    """QThread to safely read PCAP packets from SSH stdout (Wireshark-style)"""
    packet_received = pyqtSignal(dict)
    error_occurred = pyqtSignal(str)
    status_message = pyqtSignal(str)
    
    def __init__(self, process, stdout, callback=None):
        super().__init__()
        self.process = process
        self.stdout = stdout
        self.callback = callback
        self.running = False
        self.packet_counter = 0
    
    def run(self):
        """Read PCAP format packets from SSH stdout - Wireshark-compatible"""
        self.running = True
        try:
            import struct
            
            # Read PCAP global header (24 bytes) - Wireshark format
            global_header = self.stdout.read(24)
            if len(global_header) < 24:
                self.error_occurred.emit("Failed to read PCAP global header")
                return
            
            # Parse global header - check magic number
            magic_number = struct.unpack('<I', global_header[0:4])[0]
            is_native = (magic_number == 0xa1b2c3d4)  # Native byte order
            is_swapped = (magic_number == 0xd4c3b2a1)  # Swapped byte order
            
            if not (is_native or is_swapped):
                self.error_occurred.emit(f"Invalid PCAP magic number: {hex(magic_number)}")
                return
            
            # Determine byte order
            byte_order = '<' if is_native else '>'
            self.status_message.emit("PCAP header read, starting capture...")
            
            # Read packets in PCAP format (Wireshark-compatible)
            while self.running and self.process.poll() is None:
                try:
                    # Read packet header (16 bytes)
                    pkt_header = self.stdout.read(16)
                    if len(pkt_header) < 16:
                        if len(pkt_header) == 0:
                            # EOF - normal termination
                            break
                        self.error_occurred.emit(f"Incomplete packet header: {len(pkt_header)}/16")
                        break
                    
                    # Parse packet header based on byte order
                    if byte_order == '<':
                        ts_sec, ts_usec, incl_len, orig_len = struct.unpack('<IIII', pkt_header)
                    else:
                        ts_sec, ts_usec, incl_len, orig_len = struct.unpack('>IIII', pkt_header)
                    
                    timestamp = ts_sec + ts_usec / 1000000.0
                    
                    # Validate packet length
                    if incl_len > 65535 or incl_len == 0:
                        self.error_occurred.emit(f"Invalid packet length: {incl_len}")
                        break
                    
                    # Read packet data
                    packet_data = b''
                    while len(packet_data) < incl_len:
                        chunk = self.stdout.read(min(incl_len - len(packet_data), 65535))
                        if not chunk:
                            break
                        packet_data += chunk
                    
                    if len(packet_data) == incl_len:
                        # Parse packet with scapy (Wireshark-compatible parsing)
                        try:
                            scapy_packet = Ether(packet_data)
                            packet_info = parse_packet_from_scapy(
                                scapy_packet,
                                timestamp=timestamp,
                                sip_ports=DEFAULT_SIP_PORTS,
                                rtp_port_range=DEFAULT_RTP_PORT_RANGE
                            )
                            
                            if not packet_info:
                                # Fallback packet info
                                packet_info = {
                                    'timestamp': _format_packet_timestamp(timestamp),
                                    'src': '',
                                    'dst': '',
                                    'sport': 0,
                                    'dport': 0,
                                    'protocol': 'Unknown',
                                    'length': orig_len,
                                    'raw_packet': packet_data
                                }
                            
                            self.packet_counter += 1
                            packet_info['no'] = self.packet_counter
                            packet_info['frame_number'] = self.packet_counter
                            packet_info['length'] = orig_len
                            
                            # Emit packet safely via Qt signal (thread-safe)
                            self.packet_received.emit(packet_info)
                            
                            if self.packet_counter % 100 == 0:
                                self.status_message.emit(f"Captured {self.packet_counter} packets")
                                
                        except Exception as parse_error:
                            LOGGER.debug("SSH packet parse error: %s", parse_error, exc_info=True)
                            continue
                    else:
                        self.error_occurred.emit(f"Incomplete packet: {len(packet_data)}/{incl_len}")
                        break
                        
                except Exception as e:
                    if self.running:
                        LOGGER.error("SSH packet read error: %s", e, exc_info=True)
                        self.error_occurred.emit(f"Read error: {str(e)}")
                    break
            
            self.status_message.emit(f"SSH capture finished: {self.packet_counter} packets")
            
        except Exception as e:
            LOGGER.error("SSH reader thread error: %s", e, exc_info=True)
            self.error_occurred.emit(f"Thread error: {str(e)}")
        finally:
            self.running = False
    
    def stop(self):
        """Stop the reader thread"""
        self.running = False
        if self.isRunning():
            self.wait(3000)  # Wait up to 3 seconds

class RemoteAgentReceiver(QThread):
    """Thread to receive packets from remote agent"""
    packet_received = pyqtSignal(dict)
    connection_closed = pyqtSignal()
    error_occurred = pyqtSignal(str)
    
    def __init__(self, socket_conn):
        super().__init__()
        self.socket = socket_conn
        self.running = False
    
    def run(self):
        self.running = True
        try:
            while self.running:
                size_data = self.socket.recv(4)
                if not size_data or len(size_data) < 4:
                    break
                
                packet_size = struct.unpack('!I', size_data)[0]
                if packet_size == 0 or packet_size > 65535:
                    continue
                
                packet_data = b''
                while len(packet_data) < packet_size:
                    chunk = self.socket.recv(min(4096, packet_size - len(packet_data)))
                    if not chunk:
                        break
                    packet_data += chunk
                
                if len(packet_data) == packet_size:
                    try:
                        packet_info = pickle.loads(packet_data)
                        self.packet_received.emit(packet_info)
                    except Exception as e:
                        LOGGER.debug("Packet unpickle error: %s", e)
                        pass
        except Exception as e:
            if self.running:
                LOGGER.error("Receiver error: %s", e, exc_info=True)
                self.error_occurred.emit(f"Receiver error: {str(e)}")
        finally:
            self.connection_closed.emit()
            self.running = False
    
    def stop(self):
        self.running = False

class RemoteAgent:
    """Remote capture agent with Wireshark-compatible protocol support"""
    def __init__(self, host, port=2002, auth_key=None, use_rpcap=False, rpcap_interface=None, 
                 ssh_user=None, ssh_keyfile=None, ssh_password=None, capture_method='auto'):
        self.host = host
        self.port = port
        self.auth_key = auth_key
        self.use_rpcap = use_rpcap
        self.rpcap_interface = rpcap_interface
        self.ssh_user = ssh_user or 'root'
        self.ssh_keyfile = ssh_keyfile
        self.ssh_password = ssh_password
        self.capture_method = capture_method  # 'auto', 'ssh', 'rpcap', 'socket'
        self.connected = False
        self.socket = None
        self.receiver = None
        self.capture_process = None
        self.tunnel_process = None
        self.running = False
        self.rpcap_thread = None
        self.table_row = None
        self.is_capturing = False
        self.last_packet_ts = None
        self.last_status_update = 0.0
        self.ssh_capture_process = None
        self.ssh_stdout = None
        self.ssh_stderr = None
        self.ssh_reader_thread = None  # QThread for SSH packet reading
        self.tshark_reader_thread = None  # QThread for TShark packet reading
        self.active_method = None  # Track which method succeeded
        self.rpcap_client = None  # Per-agent RPCAP client instance (not global)
    
    def connect(self):
        """Original NetHawk connection method"""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.settimeout(5)
            self.socket.connect((self.host, self.port))
            if self.auth_key:
                self.socket.send(f"AUTH:{self.auth_key}\n".encode())
                response = self.socket.recv(1024).decode()
                if "OK" not in response:
                    self.socket.close()
                    return False
            self.connected = True
            return True
        except:
            return False
    
    def enhanced_connect(self):
        """Enhanced connection with multiple protocol support"""
        # First try the original connection method
        if self.connect():
            return True
            
        # If original fails, try standard protocols
        return self.try_standard_protocols()
    
    def try_standard_protocols(self):
        """Try standard remote capture protocols (Wireshark-compatible)"""
        try:
            print(f"🔍 Trying Wireshark-compatible protocols for {self.host}:{self.port}")
            
            # Determine if host is remote
            is_remote = self.host not in ['localhost', '127.0.0.1', '::1']
            
            # Method 1: Try RPCAP (if enabled and host is remote)
            if self.use_rpcap and is_remote:
                if self.try_rpcap_connection():
                    return True
            
            # Method 2: Try SSH remote capture (Wireshark sshdump-style)
            if is_remote and (self.capture_method in ['auto', 'ssh']):
                if self.try_ssh_remote_capture():
                    return True
            
            # Method 3: Try SSH tunnel (alternative SSH method)
            if is_remote and (self.capture_method in ['auto', 'ssh']):
                if self.try_ssh_tunnel():
                    return True
            
            # Method 4: Try tshark remote capture (RPCAP if enabled, local if not)
            if self.try_tshark_remote():
                return True
                
            # Method 5: Try direct socket with standard handshake (if not using RPCAP)
            if not self.use_rpcap:
                if self.try_standard_socket():
                    return True
                
            LOGGER.error("All Wireshark-compatible protocols failed for remote capture")
            return False
        except Exception as e:
            LOGGER.error(f"Standard protocol connection failed: {e}", exc_info=True)
            import traceback
            traceback.print_exc()
            return False
    
    def try_rpcap_connection(self):
        """Try RPCAP connection (Wireshark RPCAP protocol)"""
        try:
            print(f"🔍 Trying RPCAP connection to {self.host}:{self.port}...")
            
            # Create per-agent RPCAP client instance (not global - fixes race conditions)
            self.rpcap_client = RpcapClient(self.host, self.port, password=self.auth_key)
            
            if self.rpcap_client.connect():
                # List interfaces
                interfaces = self.rpcap_client.list_interfaces()
                if interfaces:
                    print(f"✅ RPCAP connected, found {len(interfaces)} interfaces")
                    if not self.rpcap_interface and interfaces:
                        # Use first non-loopback interface if available
                        for iface in interfaces:
                            if not iface.lower().startswith('lo') and iface.lower() != 'loopback':
                                self.rpcap_interface = iface
                                break
                        # Fallback to first interface if all are loopback
                        if not self.rpcap_interface:
                            self.rpcap_interface = interfaces[0]
                        print(f"📡 Using interface: {self.rpcap_interface}")
                    self.active_method = 'rpcap'
                    self.connected = True
                    return True
                else:
                    LOGGER.warning("RPCAP connected but no interfaces found")
                    self.rpcap_client.disconnect()
                    self.rpcap_client = None
                    return False
            else:
                LOGGER.error("Pure-Python RPCAP connection failed")
                self.rpcap_client = None
                return False
                
        except Exception as e:
            LOGGER.error(f"RPCAP connection error: {e}", exc_info=True)
            import traceback
            traceback.print_exc()
            if self.rpcap_client:
                try:
                    self.rpcap_client.disconnect()
                except:
                    pass
                self.rpcap_client = None
            return False
    
    def list_ssh_interfaces(self):
        """List network interfaces on remote host via SSH (Wireshark-style)"""
        try:
            # Build SSH command
            ssh_base = ['ssh']
            if self.ssh_keyfile:
                ssh_base.extend(['-i', self.ssh_keyfile])
            if self.ssh_password:
                ssh_base = ['sshpass', '-p', self.ssh_password] + ssh_base
            
            ssh_base.extend(['-o', 'StrictHostKeyChecking=no', '-o', 'ConnectTimeout=5'])
            ssh_target = f"{self.ssh_user}@{self.host}"
            
            # Try different commands to list interfaces
            interface_commands = [
                "ip -o link show | awk '{print $2}' | sed 's/:$//'",  # Linux ip command
                "ifconfig -a | grep '^[a-z]' | awk '{print $1}' | sed 's/:$//'",  # ifconfig
                "ls /sys/class/net",  # sysfs
                "netstat -i | awk 'NR>2 {print $1}'"  # netstat fallback
            ]
            
            interfaces = []
            for cmd in interface_commands:
                try:
                    full_cmd = ssh_base + [ssh_target, cmd]
                    result = subprocess.run(full_cmd, capture_output=True, timeout=5, text=True)
                    if result.returncode == 0 and result.stdout.strip():
                        interfaces = [line.strip() for line in result.stdout.strip().split('\n') if line.strip()]
                        # Filter out loopback and virtual interfaces if desired
                        interfaces = [iface for iface in interfaces if iface and not iface.startswith('lo')]
                        if interfaces:
                            print(f"✅ Found {len(interfaces)} interfaces via SSH")
                            return interfaces
                except:
                    continue
            
            print("⚠️ Could not enumerate interfaces via SSH")
            return []
            
        except Exception as e:
            print(f"❌ SSH interface enumeration failed: {e}")
            return []
    
    def try_tshark_remote(self):
        """Try tshark or native rpcap remote capture"""
        try:
            # Prefer pure-Python RPCAP if enabled
            if self.use_rpcap and self.rpcap_interface:
                print("🔍 Trying pure-Python RPCAP client (no tshark)...")
                # Use per-agent RPCAP client instance (not global)
                if not self.rpcap_client:
                    self.rpcap_client = RpcapClient(self.host, self.port, password=self.auth_key)
                if self.rpcap_client.connect():
                    self.active_method = 'rpcap'
                    self.connected = True
                    return True
                else:
                    print("❌ Pure-Python RPCAP connection failed")
                    if self.rpcap_client:
                        try:
                            self.rpcap_client.disconnect()
                        except:
                            pass
                        self.rpcap_client = None
                    return False

            print(f"🔍 Trying TShark remote capture...")
            
            # Build tshark command for capture (rpcap when enabled)
            iface_arg = 'any'
            if self.use_rpcap and self.rpcap_interface:
                # Wireshark-style RPCAP URL uses bracketed host and GUID interfaces
                iface_arg = f"rpcap://[{self.host}]:{self.port}/{self.rpcap_interface}"
                print(f"🎯 Using RPCAP interface: {iface_arg}")
            else:
                print(f"🎯 Using local interface: {iface_arg}")
                # For local capture, don't try if host is remote
                if self.host != 'localhost' and self.host != '127.0.0.1':
                    print("❌ Skipping local TShark for remote host")
                    return False

            cmd = [
                'tshark', '-i', iface_arg,
                '-T', 'fields',
                '-e', 'frame.number',
                '-e', 'frame.time',
                '-e', 'ip.src',
                '-e', 'ip.dst',
                '-e', 'ip.proto',
                '-e', 'tcp.srcport',
                '-e', 'tcp.dstport',
                '-e', 'udp.srcport',
                '-e', 'udp.dstport',
                '-e', 'frame.len',
                '-e', 'frame.protocols'
            ]
            
            print(f"🚀 Running TShark command: {' '.join(cmd)}")
            
            self.capture_process = subprocess.Popen(
                cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True,
                bufsize=1
            )
            
            # Wait a moment and check stderr for errors
            time.sleep(1)
            stderr_output = self.capture_process.stderr.read() if self.capture_process.stderr else ""
            if stderr_output:
                print(f"⚠️ TShark stderr: {stderr_output}")
            
            # Check if process started successfully
            if self.capture_process.poll() is None:
                self.connected = True
                print("✅ TShark remote capture started successfully")
                return True
            else:
                print("❌ TShark process failed to start")
                if stderr_output:
                    print(f"❌ Error details: {stderr_output}")
                return False
            
        except Exception as e:
            print(f"❌ TShark remote capture failed: {e}")
            return False
    
    def try_ssh_remote_capture(self):
        """Wireshark-style SSH remote capture (like sshdump)"""
        """Runs dumpcap/tcpdump on remote host via SSH and streams packets"""
        try:
            print(f"🔍 Trying Wireshark-style SSH remote capture to {self.host}...")
            
            # Check if dumpcap is available on remote host
            # First try dumpcap (Wireshark's capture tool), then tcpdump
            remote_capture_tools = ['dumpcap', 'tcpdump']
            remote_tool = None
            
            # Build SSH command to check for tools
            ssh_base = ['ssh']
            if self.ssh_keyfile:
                ssh_base.extend(['-i', self.ssh_keyfile])
            if self.ssh_password:
                # Use sshpass if available
                ssh_base = ['sshpass', '-p', self.ssh_password] + ssh_base
            
            ssh_base.extend(['-o', 'StrictHostKeyChecking=no', '-o', 'ConnectTimeout=5'])
            ssh_target = f"{self.ssh_user}@{self.host}"
            
            # Check which tool is available
            for tool in remote_capture_tools:
                check_cmd = ssh_base + [ssh_target, f'which {tool}']
                try:
                    result = subprocess.run(check_cmd, capture_output=True, timeout=5)
                    if result.returncode == 0:
                        remote_tool = tool
                        print(f"✅ Found {tool} on remote host")
                        break
                except:
                    continue
            
            if not remote_tool:
                print("❌ No capture tool (dumpcap/tcpdump) found on remote host")
                return False
            
            # Build remote capture command
            # Use dumpcap if available (better for Wireshark compatibility)
            if remote_tool == 'dumpcap':
                # dumpcap outputs pcap format directly
                remote_cmd = [
                    'dumpcap', '-i', 'any',  # Capture from any interface
                    '-w', '-',  # Write to stdout
                    '-q',  # Quiet mode
                    '-f', 'not port 22'  # Exclude SSH traffic
                ]
            else:
                # tcpdump fallback
                remote_cmd = [
                    'tcpdump', '-i', 'any',
                    '-U',  # Unbuffered output
                    '-w', '-',  # Write to stdout
                    '-n',  # Don't resolve names
                    'not port 22'  # Exclude SSH traffic
                ]
            
            # Build full SSH command
            ssh_cmd = ssh_base + [ssh_target] + remote_cmd
            
            print(f"🚀 Starting SSH remote capture: {' '.join(ssh_cmd[:5])}... {remote_tool}")
            
            # Start SSH process with remote capture
            self.ssh_capture_process = subprocess.Popen(
                ssh_cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                bufsize=0  # Unbuffered
            )
            
            # Wait longer for SSH connection to establish (increased from 0.5s to 2s)
            time.sleep(2)
            if self.ssh_capture_process.poll() is not None:
                stderr = self.ssh_capture_process.stderr.read(1024).decode('utf-8', errors='ignore')
                print(f"❌ SSH capture process failed: {stderr}")
                return False
            
            self.ssh_stdout = self.ssh_capture_process.stdout
            self.ssh_stderr = self.ssh_capture_process.stderr
            self.active_method = 'ssh'
            self.connected = True
            print("✅ SSH remote capture started successfully")
            return True
            
        except Exception as e:
            print(f"❌ SSH remote capture failed: {e}")
            import traceback
            traceback.print_exc()
            return False
    
    def try_ssh_tunnel(self):
        """Try SSH tunnel for remote capture (alternative method)"""
        try:
            print(f"🔍 Trying SSH tunnel to {self.host}:{self.port}...")
            
            # Build SSH tunnel command
            tunnel_cmd = ['ssh']
            if self.ssh_keyfile:
                tunnel_cmd.extend(['-i', self.ssh_keyfile])
            if self.ssh_password:
                tunnel_cmd = ['sshpass', '-p', self.ssh_password] + tunnel_cmd
            
            tunnel_cmd.extend([
                '-L', f'2002:{self.host}:{self.port}',
                '-o', 'StrictHostKeyChecking=no',
                '-o', 'ConnectTimeout=5',
                '-N',  # No command execution
                f'{self.ssh_user}@{self.host}'
            ])
            
            print(f"🚀 Creating SSH tunnel: {' '.join(tunnel_cmd[:3])}...")
            self.tunnel_process = subprocess.Popen(
                tunnel_cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE
            )
            
            # Wait for tunnel to establish
            time.sleep(2)
            
            # Check if tunnel process is still running
            if self.tunnel_process.poll() is not None:
                stderr = self.tunnel_process.stderr.read(1024).decode('utf-8', errors='ignore')
                print(f"❌ SSH tunnel failed: {stderr}")
                return False
            
            # Connect to local tunnel endpoint
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.settimeout(5)
            self.socket.connect(('localhost', 2002))
            
            self.active_method = 'ssh_tunnel'
            self.connected = True
            print("✅ SSH tunnel established successfully")
            return True
            
        except Exception as e:
            print(f"❌ SSH tunnel failed: {e}")
            if self.tunnel_process:
                self.tunnel_process.terminate()
            return False
    
    def try_standard_socket(self):
        """Try standard socket with different handshake methods"""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.settimeout(5)
            self.socket.connect((self.host, self.port))
            
            # Try different handshake methods
            handshake_methods = [
                self.try_netHawk_handshake,
                self.try_standard_handshake,
                self.try_wireshark_handshake
            ]
            
            for method in handshake_methods:
                if method():
                    return True
                    
            return False
            
        except Exception as e:
            print(f"Standard socket failed: {e}")
            return False
    
    def try_netHawk_handshake(self):
        """Try NetHawk custom protocol handshake"""
        try:
            if self.auth_key:
                self.socket.send(f"AUTH:{self.auth_key}\n".encode())
                response = self.socket.recv(1024).decode()
                return "OK" in response
            return True
        except:
            return False
    
    def try_standard_handshake(self):
        """Try standard remote capture handshake"""
        try:
            # Send standard start command
            self.socket.send(b"START\n")
            response = self.socket.recv(1024).decode()
            return "OK" in response or "READY" in response
        except:
            return False
    
    def try_wireshark_handshake(self):
        """Try Wireshark-compatible handshake"""
        try:
            # Send Wireshark-style command
            self.socket.send(b"CAPTURE\n")
            response = self.socket.recv(1024).decode()
            return "OK" in response or "READY" in response
        except:
            return False
    
    def start_remote_capture(self, filters=None, packet_callback=None):
        """Original NetHawk capture method"""
        if not self.connected:
            return False
        try:
            cmd = "START_CAPTURE"
            if filters:
                cmd += f":{filters}"
            self.socket.send(f"{cmd}\n".encode())
            self.socket.settimeout(None)
            
            if packet_callback:
                self.receiver = RemoteAgentReceiver(self.socket)
                self.receiver.packet_received.connect(packet_callback)
                self.receiver.start()
            
            return True
        except:
            return False
    
    def start_enhanced_capture(self, filters=None, packet_callback=None):
        """Enhanced capture with Wireshark-compatible protocol support"""
        if not self.connected:
            LOGGER.error("Cannot start capture: not connected to remote host")
            return False
            
        try:
            LOGGER.info(f"Starting enhanced capture (active_method={self.active_method}, host={self.host}, port={self.port})")
            
            # Route to appropriate capture method based on active_method
            if self.active_method == 'ssh':
                # SSH remote capture (Wireshark sshdump-style)
                LOGGER.info("Using SSH remote capture method")
                return self.start_ssh_capture(filters, packet_callback)
            elif self.active_method == 'rpcap':
                # RPCAP capture - use per-agent client instance
                if self.rpcap_client and self.rpcap_interface:
                    LOGGER.info(f"Starting RPCAP capture on interface '{self.rpcap_interface}' (host={self.host}, port={self.port})")
                    if self.rpcap_client.start_capture(self.rpcap_interface, packet_callback):
                        LOGGER.info("RPCAP capture started successfully")
                        self.running = True
                        return True
                    else:
                        LOGGER.error("RPCAP capture failed to start")
                        return False
                else:
                    LOGGER.error(f"RPCAP capture cannot start: rpcap_client={self.rpcap_client is not None}, rpcap_interface={self.rpcap_interface}")
                    return False
            elif self.rpcap_thread:
                LOGGER.info(f"Starting native RPCAP capture on {self.rpcap_interface}")
                self.running = True
                if packet_callback:
                    try:
                        self.rpcap_thread.packet_received.disconnect()
                    except TypeError:
                        pass
                    self.rpcap_thread.packet_received.connect(packet_callback)
                    self.rpcap_thread.set_callback(None)
                else:
                    self.rpcap_thread.set_callback(None)
                self.rpcap_thread.start()
                return True
            elif self.capture_process:
                # Handle tshark process
                LOGGER.info("Using TShark capture method")
                return self.start_tshark_capture(filters, packet_callback)
            elif self.socket:
                # Handle socket connection (original functionality or SSH tunnel)
                LOGGER.info("Using socket-based capture method")
                return self.start_remote_capture(filters, packet_callback)
            else:
                LOGGER.error("No active capture method available")
                return False
                
        except Exception as e:
            print(f"Enhanced capture failed: {e}")
            import traceback
            traceback.print_exc()
            return False

class RpcapCaptureThread(QThread):
    """Native RPCAP capture using python-libpcap/pcapy when available"""
    packet_received = pyqtSignal(dict)
    error_occurred = pyqtSignal(str)
    status_changed = pyqtSignal(str)

    def __init__(self, host, port, interface, packet_callback=None):
        super().__init__()
        self.host = host
        self.port = port
        self.interface = interface  # can be GUID like {..}
        self.packet_callback = packet_callback
        self._running = False
        self._pcap = None
        self._packet_no = 0

    def set_callback(self, cb):
        self.packet_callback = cb

    def run(self):
        if not PCAP_AVAILABLE:
            self.error_occurred.emit('libpcap bindings not available (install python-libpcap or pcapy)')
            return

        try:
            self._running = True
            self.status_changed.emit('Connecting RPCAP...')

            # Build rpcap source string; many libpcap bindings accept rpcap://host:port/iface
            source = f"rpcap://{self.host}:{self.port}/{self.interface}"

            if PCAP_LIB == 'pcap':
                # python-libpcap
                import pcap
                # open_live allows rpcap sources per libpcap
                self._pcap = pcap.pcap(name=source, promisc=True, immediate=True, timeout_ms=1000)
                self.status_changed.emit('RPCAP connected')
                for ts, pkt in self._pcap:
                    if not self._running:
                        break
                    self._emit_packet(ts, pkt)
            else:
                # pcapy fallback
                import pcapy
                self._pcap = pcapy.open_live(source, 65535, 1, 1000)
                self.status_changed.emit('RPCAP connected')
                while self._running:
                    header, pkt = self._pcap.next()
                    if not header:  # timeout
                        continue
                    ts = header.getts()[0] + header.getts()[1] / 1_000_000
                    self._emit_packet(ts, pkt)

        except Exception as e:
            self.error_occurred.emit(f'RPCAP error: {str(e)}')
        finally:
            self._running = False
            try:
                if self._pcap and hasattr(self._pcap, 'close'):
                    self._pcap.close()
            except:
                pass

    def stop(self):
        self._running = False

    def _emit_packet(self, ts, raw_bytes):
        # Convert to NetHawk packet_info dict (minimal fields; downstream enriches)
        try:
            self._packet_no += 1
            packet_info = None
            try:
                scapy_packet = Ether(raw_bytes)
                packet_info = parse_packet_from_scapy(
                    scapy_packet,
                    timestamp=ts,
                    sip_ports=DEFAULT_SIP_PORTS,
                    rtp_port_range=DEFAULT_RTP_PORT_RANGE
                )
            except Exception as parse_error:
                LOGGER.debug("RPCAP (libpcap) parse error: %s", parse_error, exc_info=True)

            if not packet_info:
                packet_info = {
                    'timestamp': _format_packet_timestamp(ts),
                    'src': '',
                    'dst': '',
                    'sport': 0,
                    'dport': 0,
                    'protocol': 'Unknown',
                    'length': len(raw_bytes),
                    'qos_dscp': 0,
                    'qos_name': 'BE',
                    'flags': '',
                    'info': 'RPCAP packet'
                }

            packet_info['no'] = self._packet_no
            packet_info['frame_number'] = self._packet_no
            packet_info['length'] = len(raw_bytes)
            packet_info['raw_packet'] = raw_bytes

            if self.packet_callback:
                self.packet_callback(packet_info)
            self.packet_received.emit(packet_info)
        except Exception as emit_error:
            LOGGER.error("Failed to emit RPCAP packet: %s", emit_error, exc_info=True)
    
    
    def stop_capture(self):
        """Stop remote capture"""
        self.running = False
        self.is_capturing = False
        
        # Stop SSH capture process and reader thread
        if self.ssh_reader_thread:
            try:
                self.ssh_reader_thread.stop()
                if self.ssh_reader_thread.isRunning():
                    self.ssh_reader_thread.wait(3000)
            except Exception as e:
                LOGGER.debug("Error stopping SSH reader thread: %s", e)
            self.ssh_reader_thread = None
        
        if self.ssh_capture_process:
            try:
                self.ssh_capture_process.terminate()
                self.ssh_capture_process.wait(timeout=2)
            except:
                try:
                    self.ssh_capture_process.kill()
                except:
                    pass
            self.ssh_capture_process = None
            self.ssh_stdout = None
            self.ssh_stderr = None
        
        if self.receiver:
            self.receiver.stop()
            self.receiver.wait(1000)
            self.receiver = None
        
        if self.rpcap_thread and self.rpcap_thread.isRunning():
            self.rpcap_thread.stop()
            self.rpcap_thread.wait(1000)
            self.rpcap_thread = None
        
        # Stop TShark reader thread
        if self.tshark_reader_thread:
            try:
                self.tshark_reader_thread.stop()
                if self.tshark_reader_thread.isRunning():
                    self.tshark_reader_thread.wait(3000)
            except Exception as e:
                LOGGER.debug("Error stopping TShark reader thread: %s", e)
            self.tshark_reader_thread = None
        
        if self.capture_process:
            self.capture_process.terminate()
            try:
                self.capture_process.wait(timeout=2)
            except:
                try:
                    self.capture_process.kill()
                except:
                    pass
            self.capture_process = None
        
        if self.tunnel_process:
            self.tunnel_process.terminate()
            try:
                self.tunnel_process.wait(timeout=2)
            except:
                try:
                    self.tunnel_process.kill()
                except:
                    pass
            self.tunnel_process = None
        
        # Stop per-agent RPCAP client (not global)
        if self.rpcap_client:
            try:
                self.rpcap_client.stop_capture()
                self.rpcap_client.disconnect()
            except Exception as e:
                LOGGER.debug("Error stopping RPCAP client: %s", e)
            self.rpcap_client = None
        
        if self.connected and self.socket:
            try:
                self.socket.send(b"STOP_CAPTURE\n")
            except:
                pass
    
    def disconnect(self):
        """Disconnect from remote server"""
        self.stop_capture()
        if self.socket:
            try:
                self.socket.close()
            except:
                pass
            self.socket = None  # FIXED: Set to None after close
        if self.rpcap_client:
            try:
                self.rpcap_client.disconnect()
            except:
                pass
            self.rpcap_client = None
        self.connected = False

class PCAPLoader(QThread):
    """THREAD-SAFE PCAP loader"""
    packet_batch_loaded = pyqtSignal(list)
    loading_progress = pyqtSignal(int, int)
    loading_complete = pyqtSignal(int)
    loading_error = pyqtSignal(str)
    
    def __init__(self, filename, max_packets=None):
        super().__init__()
        self.filename = filename
        self.max_packets = max_packets or MAX_PACKETS
        self.should_stop = False
        self._running = False
    
    def run(self):
        self._running = True
        try:
            self.loading_progress.emit(0, 0)
            
            # Try PcapReader first (memory-efficient streaming), fallback to rdpcap if it fails
            use_pcap_reader = True
            total = 0
            skip_count = 0
            
            try:
                from scapy.utils import PcapReader
                
                # First, count total packets (for progress bar)
                try:
                    reader = PcapReader(self.filename)
                    total = 0
                    try:
                        for _ in reader:
                            total += 1
                            if total > self.max_packets * 2:  # Safety limit for counting
                                break
                    except Exception as count_iter_error:
                        LOGGER.debug("Error during packet counting: %s", count_iter_error)
                    finally:
                        try:
                            reader.close()
                        except:
                            pass
                    
                    # If file is too large, we'll only load the last max_packets
                    skip_count = max(0, total - self.max_packets) if total > self.max_packets else 0
                    total = min(total, self.max_packets)
                except Exception as count_error:
                    # If counting fails, try to load anyway
                    LOGGER.warning("Could not count packets with PcapReader: %s", count_error)
                    total = 0
                    skip_count = 0
            except ImportError:
                # PcapReader not available, use rdpcap fallback
                LOGGER.warning("PcapReader not available, using rdpcap fallback")
                use_pcap_reader = False
            
            if self.should_stop:
                return
            
            self.loading_progress.emit(0, total if total > 0 else 1)
            
            # Now load packets in batches
            batch_size = 100
            batch = []
            packet_count = 0
            skipped = 0
            
            try:
                if use_pcap_reader:
                    # Use PcapReader for streaming
                    try:
                        reader = PcapReader(self.filename)
                        try:
                            for packet in reader:
                                if self.should_stop:
                                    break
                                
                                # Skip packets if we're only loading the last N
                                if skip_count > 0 and skipped < skip_count:
                                    skipped += 1
                                    continue
                                
                                try:
                                    packet_info = self.parse_pcap_packet(packet, packet_count + 1)
                                    if packet_info:
                                        batch.append(packet_info)
                                        packet_count += 1
                                    
                                    if len(batch) >= batch_size:
                                        if not self.should_stop:
                                            self.packet_batch_loaded.emit(batch)
                                        batch = []
                                        if total > 0:
                                            self.loading_progress.emit(packet_count, total)
                                        else:
                                            self.loading_progress.emit(packet_count, packet_count)
                                        self.msleep(10)  # Yield to GUI thread
                                    
                                    # Stop if we've loaded enough
                                    if packet_count >= self.max_packets:
                                        break
                                except Exception as parse_error:
                                    # Skip malformed packets instead of crashing
                                    LOGGER.debug("Error parsing packet %d: %s", packet_count + 1, parse_error)
                                    continue
                        finally:
                            try:
                                reader.close()
                            except:
                                pass
                    except Exception as pcap_reader_error:
                        LOGGER.warning("PcapReader failed, falling back to rdpcap: %s", pcap_reader_error)
                        use_pcap_reader = False
                
                # Fallback to rdpcap if PcapReader failed or not available
                if not use_pcap_reader:
                    from scapy.all import rdpcap
                    LOGGER.info("Loading PCAP with rdpcap (may use more memory for large files)")
                    packets = rdpcap(self.filename)
                    total = len(packets)
                    
                    if total > self.max_packets:
                        packets = packets[-self.max_packets:]
                        total = self.max_packets
                    
                    self.loading_progress.emit(0, total)
                    
                    for idx, packet in enumerate(packets):
                        if self.should_stop:
                            break
                        
                        try:
                            packet_info = self.parse_pcap_packet(packet, idx + 1)
                            if packet_info:
                                batch.append(packet_info)
                                packet_count += 1
                            
                            if len(batch) >= batch_size:
                                if not self.should_stop:
                                    self.packet_batch_loaded.emit(batch)
                                batch = []
                                self.loading_progress.emit(idx + 1, total)
                                self.msleep(10)  # Yield to GUI thread
                        except Exception as parse_error:
                            # Skip malformed packets instead of crashing
                            LOGGER.debug("Error parsing packet %d: %s", idx + 1, parse_error)
                            continue
                    
                    # Update packet_count for rdpcap fallback
                    if batch:
                        packet_count = len([p for p in batch if p])
                    else:
                        packet_count = 0
                    
            except Exception as read_error:
                if not self.should_stop:
                    error_msg = f"Failed to read PCAP file: {str(read_error)}"
                    LOGGER.error("PCAP read error: %s", error_msg, exc_info=True)
                    self.loading_error.emit(error_msg)
                return
            
            # Emit remaining batch
            if batch and not self.should_stop:
                try:
                    self.packet_batch_loaded.emit(batch)
                except Exception as emit_error:
                    LOGGER.error("Error emitting final batch: %s", emit_error, exc_info=True)
            
            if not self.should_stop:
                final_total = total if total > 0 else packet_count
                self.loading_progress.emit(packet_count, final_total)
                self.loading_complete.emit(packet_count)
        except Exception as e:
            if not self.should_stop:
                error_msg = f"Failed to load PCAP: {str(e)}"
                LOGGER.error("PCAP load error: %s", error_msg, exc_info=True)
                self.loading_error.emit(error_msg)
        finally:
            self._running = False
    
    def stop(self):
        self.should_stop = True
    
    def is_running(self):
        return self._running
    
    def parse_pcap_packet(self, packet, packet_no):
        try:
            # Safely extract timestamp with error handling
            try:
                if hasattr(packet, 'time'):
                    timestamp = datetime.fromtimestamp(float(packet.time)).strftime('%H:%M:%S.%f')[:-3]
                else:
                    timestamp = datetime.now().strftime('%H:%M:%S.%f')[:-3]
            except (ValueError, TypeError, OSError, OverflowError) as ts_error:
                # Handle invalid timestamps gracefully
                LOGGER.debug("Invalid timestamp in packet %d: %s", packet_no, ts_error)
                timestamp = datetime.now().strftime('%H:%M:%S.%f')[:-3]
            
            # Safely get packet length
            try:
                packet_len = len(packet) if packet else 0
            except:
                packet_len = 0
            
            # Safely extract raw packet bytes (limit size to prevent memory issues)
            raw_packet = None
            try:
                if packet and packet_len < 2000:
                    raw_packet = bytes(packet)
            except Exception as bytes_error:
                LOGGER.debug("Could not extract raw packet bytes: %s", bytes_error)
            
            packet_info = {
                'no': packet_no,
                'timestamp': timestamp,
                'src': 'Unknown', 'dst': 'Unknown',
                'sport': 0, 'dport': 0,
                'protocol': 'Other',
                'length': packet_len,
                'qos_dscp': 0, 'qos_name': 'BE',
                'flags': '', 'info': '',
                'raw_packet': raw_packet
            }
            
            # Safely check for ARP layer
            try:
                if packet.haslayer(ARP):
                    arp = packet[ARP]
                    packet_info['protocol'] = 'ARP'
                    packet_info['src'] = getattr(arp, 'psrc', 'Unknown')
                    packet_info['dst'] = getattr(arp, 'pdst', 'Unknown')
                    packet_info['info'] = f"Who has {packet_info['dst']}? Tell {packet_info['src']}"
                    return packet_info
            except Exception as arp_error:
                LOGGER.debug("Error parsing ARP layer: %s", arp_error)
            
            # Safely check for IP layer
            try:
                if not packet.haslayer(IP):
                    return None
            except Exception:
                return None
            
            # Safely extract IP information
            try:
                ip = packet[IP]
                packet_info['src'] = getattr(ip, 'src', 'Unknown')
                packet_info['dst'] = getattr(ip, 'dst', 'Unknown')
                tos = getattr(ip, 'tos', 0)
                packet_info['qos_dscp'] = (tos >> 2) & 0x3F
                packet_info['qos_name'] = DSCP_CLASSES.get(packet_info['qos_dscp'], ('', 'UK', '#78909C'))[1]
            except Exception as ip_error:
                LOGGER.debug("Error parsing IP layer: %s", ip_error)
                return None
            
            # Safely parse TCP layer
            try:
                if packet.haslayer(TCP):
                    tcp = packet[TCP]
                    packet_info['sport'] = getattr(tcp, 'sport', 0)
                    packet_info['dport'] = getattr(tcp, 'dport', 0)
                    packet_info['protocol'] = self.detect_protocol(packet_info['sport'], packet_info['dport'], packet, 'TCP')
                    flags = []
                    try:
                        if tcp.flags.F: flags.append('FIN')
                        if tcp.flags.S: flags.append('SYN')
                        if tcp.flags.R: flags.append('RST')
                        if tcp.flags.P: flags.append('PSH')
                        if tcp.flags.A: flags.append('ACK')
                    except:
                        pass
                    packet_info['flags'] = ','.join(flags)
                    seq = getattr(tcp, 'seq', 0)
                    packet_info['info'] = f"{packet_info['protocol']} [{packet_info['flags']}] Seq={seq}"
            except Exception as tcp_error:
                LOGGER.debug("Error parsing TCP layer: %s", tcp_error)
            
            # Safely parse UDP layer
            try:
                if packet.haslayer(UDP):
                    udp = packet[UDP]
                    packet_info['sport'] = getattr(udp, 'sport', 0)
                    packet_info['dport'] = getattr(udp, 'dport', 0)
                    packet_info['protocol'] = self.detect_protocol(packet_info['sport'], packet_info['dport'], packet, 'UDP')
                    packet_info['info'] = f"{packet_info['protocol']} {packet_info['sport']}→{packet_info['dport']}"
            except Exception as udp_error:
                LOGGER.debug("Error parsing UDP layer: %s", udp_error)
            
            # Safely parse ICMP layer
            try:
                if packet.haslayer(ICMP):
                    icmp = packet[ICMP]
                    packet_info['protocol'] = 'ICMP'
                    icmp_type = getattr(icmp, 'type', 0)
                    packet_info['info'] = f"ICMP Type {icmp_type}"
            except Exception as icmp_error:
                LOGGER.debug("Error parsing ICMP layer: %s", icmp_error)
            
            return packet_info
        except:
            return None
    
    def detect_protocol(self, sport, dport, packet, transport):
        port_map = {20: 'FTP-DATA', 21: 'FTP', 22: 'SSH', 25: 'SMTP', 53: 'DNS',
                    80: 'HTTP', 443: 'HTTPS', 5060: 'SIP'}
        if sport in port_map:
            return port_map[sport]
        if dport in port_map:
            return port_map[dport]
        
        if sport in [5060, 5061] or dport in [5060, 5061]:
            if packet.haslayer(Raw):
                payload = bytes(packet[Raw].load)
                try:
                    payload_str = payload.decode('utf-8', errors='ignore')[:200]
                    if 'SIP/2.0' in payload_str or 'INVITE sip:' in payload_str or 'REGISTER sip:' in payload_str:
                        return 'SIP'
                except:
                    pass
        
        if transport == 'UDP' and (10000 <= sport <= 20000 or 10000 <= dport <= 20000):
            if packet.haslayer(Raw):
                payload = bytes(packet[Raw].load)
                if len(payload) >= 12:
                    version = (payload[0] >> 6) & 0x03
                    if version == 2:
                        payload_type = payload[1] & 0x7F
                        if payload_type <= 127:
                            return 'RTP'
        
        if packet.haslayer(Raw):
            payload = bytes(packet[Raw].load)
            try:
                payload_str = payload.decode('utf-8', errors='ignore')[:50]
                if any(m in payload_str for m in ['GET ', 'POST ', 'HTTP/']):
                    return 'HTTP'
            except:
                pass
        
        return transport

class NetworkCapture(QThread):
    """Network capture thread"""
    packet_received = pyqtSignal(dict)
    sip_message_received = pyqtSignal(dict)
    rtp_packet_received = pyqtSignal(dict)
    status_changed = pyqtSignal(str)
    error_occurred = pyqtSignal(str)
    
    def __init__(self):
        super().__init__()
        self.running = False
        self.packet_count = 0
        self.interface = None
        self.capture_filter = 'all'
        self.sip_ports = list(DEFAULT_SIP_PORTS)
        self.rtp_port_range = DEFAULT_RTP_PORT_RANGE
    
    def start_capture(self, interface=None, capture_filter='all'):
        self.interface = interface
        self.capture_filter = capture_filter
        self.running = True
        self.start()
    
    def run(self):
        try:
            self.status_changed.emit("Starting capture...")
            bpf_filter = None if self.capture_filter == 'all' else None
            sniff(iface=self.interface, filter=bpf_filter, prn=self.packet_handler,
                  stop_filter=lambda x: not self.running, store=False)
        except Exception as e:
            self.error_occurred.emit(f"Capture failed: {str(e)}")
    
    def packet_handler(self, packet):
        try:
            packet_info = self.parse_packet(packet)
            if packet_info:
                self.packet_count += 1
                packet_info['no'] = self.packet_count
                packet_info['raw_packet'] = bytes(packet)
                self.packet_received.emit(packet_info)
                
                if packet_info['protocol'] == 'SIP':
                    self.parse_sip_packet(packet)
                elif packet_info['protocol'] == 'RTP':
                    self.parse_rtp_packet(packet)
        except:
            pass
    
    def parse_packet(self, packet):
        try:
            return parse_packet_from_scapy(
                packet,
                sip_ports=self.sip_ports,
                rtp_port_range=self.rtp_port_range
            )
        except Exception:
            return None
    
    def detect_app_protocol(self, sport, dport, packet, transport):
        return detect_application_protocol(
            packet,
            sport,
            dport,
            transport,
            sip_ports=self.sip_ports,
            rtp_port_range=self.rtp_port_range
        )
    
    def parse_sip_packet(self, packet):
        try:
            if not packet.haslayer(Raw):
                return
            raw_data = bytes(packet[Raw].load)
            sip_data = raw_data.decode('utf-8', errors='ignore')
            lines = sip_data.split('\r\n')
            if not lines:
                return
            
            sip_info = {
                'timestamp': time.time(),
                'src': packet[IP].src, 'dst': packet[IP].dst,
                'sport': packet[UDP].sport, 'dport': packet[UDP].dport,
                'raw': sip_data, 'headers': {}
            }
            
            first_line = lines[0]
            if first_line.startswith('SIP/'):
                parts = first_line.split(' ', 2)
                sip_info['type'] = 'response'
                sip_info['status_code'] = parts[1] if len(parts) > 1 else 'Unknown'
                sip_info['method'] = sip_info['status_code']
            else:
                parts = first_line.split(' ')
                sip_info['type'] = 'request'
                sip_info['method'] = parts[0] if parts else 'Unknown'
            
            for line in lines[1:]:
                if ':' in line:
                    key, value = line.split(':', 1)
                    sip_info['headers'][key.strip()] = value.strip()
            
            sip_info['call_id'] = sip_info['headers'].get('Call-ID', '')
            sip_info['from'] = sip_info['headers'].get('From', '')
            sip_info['to'] = sip_info['headers'].get('To', '')
            self.sip_message_received.emit(sip_info)
        except:
            pass
    
    def parse_rtp_packet(self, packet):
        try:
            if not packet.haslayer(Raw):
                return
            payload = bytes(packet[Raw].load)
            if len(payload) < 12:
                return
            
            version = (payload[0] >> 6) & 0x03
            if version != 2:
                return
            
            cc = payload[0] & 0x0F
            payload_type = payload[1] & 0x7F
            sequence = struct.unpack('!H', payload[2:4])[0]
            timestamp = struct.unpack('!I', payload[4:8])[0]
            ssrc = struct.unpack('!I', payload[8:12])[0]
            
            header_len = 12 + (cc * 4)
            rtp_payload = payload[header_len:]
            
            rtp_info = {
                'timestamp': time.time(),
                'src': packet[IP].src, 'dst': packet[IP].dst,
                'sport': packet[UDP].sport, 'dport': packet[UDP].dport,
                'ssrc': ssrc, 'sequence': sequence,
                'rtp_timestamp': timestamp,
                'payload_type': payload_type,
                'payload': rtp_payload
            }
            self.rtp_packet_received.emit(rtp_info)
        except:
            pass
    
    def stop_capture(self):
        self.running = False

class ModernPacketTable(QTableWidget):

    packet_selected = pyqtSignal(dict)
    
    def __init__(self):
        super().__init__()

    def setup_table(self):

        header.setSectionResizeMode(6, QHeaderView.ResizeToContents)
        header.setSectionResizeMode(7, QHeaderView.Stretch)
    
    def add_packet(self, packet_info):
        with self._pending_lock:
            if len(self.packets) >= self.max_packets:
                self.packets.pop(0)
                if self.filtered_packets:
                    self.filtered_packets.pop(0)
            
            self.packets.append(packet_info)
            
            if self.should_display(packet_info):
                display_info = {k: v for k, v in packet_info.items() if k != 'raw_packet'}
                self.filtered_packets.append(display_info)
                self._pending_packets.append(display_info)
    
    def flush_pending_packets(self):
        with self._pending_lock:
            if not self._pending_packets:
                return
            
            batch = self._pending_packets[:]
            self._pending_packets.clear()
        
        self.setUpdatesEnabled(False)
        try:
            while self.rowCount() >= self.max_packets:
                self.removeRow(0)
            
            for packet_info in batch:
                self.display_packet_fast(packet_info)
        finally:
            self.setUpdatesEnabled(True)
            scrollbar = self.verticalScrollBar()
            if scrollbar.value() >= scrollbar.maximum() - 10:
                self.scrollToBottom()
    
    def display_packet_fast(self, packet_info):
        row = self.rowCount()
        self.insertRow(row)
        
        no_item = QTableWidgetItem(str(packet_info['no']))
        no_item.setTextAlignment(Qt.AlignCenter)
        self.setItem(row, 0, no_item)
        
        time_item = QTableWidgetItem(packet_info['timestamp'])
        time_item.setFont(QFont("Consolas", 8))
        self.setItem(row, 1, time_item)
        
        self.setItem(row, 2, QTableWidgetItem(packet_info['src']))
        self.setItem(row, 3, QTableWidgetItem(packet_info['dst']))
        
        protocol_item = QTableWidgetItem(packet_info['protocol'])
        protocol_item.setTextAlignment(Qt.AlignCenter)
        protocol_item.setFont(QFont("Arial", 8, QFont.Bold))
        color = QColor(PROTOCOL_COLORS.get(packet_info['protocol'], PROTOCOL_COLORS['Other']))
        protocol_item.setBackground(QBrush(color))
        protocol_item.setForeground(QBrush(QColor('white')))
        self.setItem(row, 4, protocol_item)
        
        qos_item = QTableWidgetItem(packet_info.get('qos_name', 'BE'))
        qos_item.setTextAlignment(Qt.AlignCenter)
        qos_color = QColor(DSCP_CLASSES.get(packet_info.get('qos_dscp', 0), ('', '', '#78909C'))[2])
        qos_item.setBackground(QBrush(qos_color))
        qos_item.setForeground(QBrush(QColor('white')))
        self.setItem(row, 5, qos_item)
        
        len_item = QTableWidgetItem(str(packet_info['length']))
        len_item.setTextAlignment(Qt.AlignRight | Qt.AlignVCenter)
        self.setItem(row, 6, len_item)
        
        info = packet_info.get('info', '')
        if len(info) > 100:
            info = info[:97] + '...'
        self.setItem(row, 7, QTableWidgetItem(info))
    
    def should_display(self, packet_info):
        """Use advanced filter parser"""
        if not self.filter_text:
            return True
        
        try:
            return FilterParser.parse_and_evaluate(self.filter_text, packet_info)
        except:
            filter_lower = self.filter_text.lower()
            if packet_info['protocol'].lower() == filter_lower:
                return True
            if filter_lower in packet_info['src'].lower() or filter_lower in packet_info['dst'].lower():
                return True
            if filter_lower in packet_info.get('info', '').lower():
                return True
            return False
    
    def apply_filter(self, filter_text):
        """Apply advanced filter"""
        self.filter_text = filter_text
        self.filtered_packets.clear()
        
        self.setUpdatesEnabled(False)
        self.setRowCount(0)
        try:
            for packet in self.packets:
                if self.should_display(packet):
                    display_info = {k: v for k, v in packet.items() if k != 'raw_packet'}
                    self.filtered_packets.append(display_info)
                    self.display_packet_fast(display_info)
        finally:
            self.setUpdatesEnabled(True)
    
    def on_selection(self):
        row = self.currentRow()
        if 0 <= row < len(self.filtered_packets):
            packet_no = self.filtered_packets[row]['no']
            full_packet = next((p for p in self.packets if p['no'] == packet_no), None)
            if full_packet:
                self.packet_selected.emit(full_packet)
    
    def clear_all(self):
        with self._pending_lock:
            self.packets.clear()
            self.filtered_packets.clear()
            self._pending_packets.clear()
        
        self.setUpdatesEnabled(False)
        try:
            self.setRowCount(0)
        finally:
            self.setUpdatesEnabled(True)

    def format_hex_dump(self, data):
        if not data:
            return "No raw data available"
        if len(data) > 256:
            data = data[:256]
            truncated = True
        else:
            truncated = False
        
        lines = []
        for i in range(0, len(data), 16):
            chunk = data[i:i+16]
            hex_part = ' '.join(f'{b:02x}' for b in chunk)
            hex_part = hex_part.ljust(48)
            ascii_part = ''.join(chr(b) if 32 <= b < 127 else '.' for b in chunk)
            lines.append(f'{i:04x}  {hex_part}  {ascii_part}')
        
        result = '\n'.join(lines)
        if truncated:
            result += '\n... (truncated)'
        return result

class CallFlowDiagram(QGraphicsView):
    """Compact SIP call flow diagram"""
    message_selected = pyqtSignal(dict)
    
    def __init__(self):
        super().__init__()
        self.scene = QGraphicsScene()
        self.setScene(self.scene)
        self.setRenderHint(QPainter.Antialiasing)
        self.setStyleSheet(f"QGraphicsView {{ background-color: {COLORS['dark']}; border: none; }}")
    
    def draw_call_flow(self, sip_call):
        self.scene.clear()
        
        if not sip_call.messages:
            text = self.scene.addText("No SIP messages", QFont("Arial", 10))
            text.setDefaultTextColor(QColor(COLORS['light']))
            text.setPos(50, 50)
            return
        
        endpoints = set()
        for msg in sip_call.messages:
            endpoints.add(msg['src'])
            endpoints.add(msg['dst'])
        endpoints = sorted(list(endpoints))
        
        if len(endpoints) < 2:
            return
        
        spacing = 250
        start_y = 60
        msg_spacing = 40
        
        positions = {}
        for i, endpoint in enumerate(endpoints):
            x = i * spacing + 120
            positions[endpoint] = x
            
            rect = QGraphicsRectItem(x-60, start_y, 120, 30)
            rect.setBrush(QBrush(QColor(COLORS['card'])))
            rect.setPen(QPen(QColor(COLORS['primary']), 2))
            self.scene.addItem(rect)
            
            text = self.scene.addText(endpoint, QFont("Arial", 8, QFont.Bold))
            text.setDefaultTextColor(QColor(COLORS['light']))
            text.setPos(x - text.boundingRect().width()/2, start_y + 8)
            
            line = QGraphicsLineItem(x, start_y + 30, x, start_y + 30 + len(sip_call.messages) * msg_spacing + 30)
            line.setPen(QPen(QColor(COLORS['card']), 2, Qt.DashLine))
            self.scene.addItem(line)
        
        y = start_y + 60
        for msg in sip_call.messages:
            src_x = positions.get(msg['src'], positions[endpoints[0]])
            dst_x = positions.get(msg['dst'], positions[endpoints[-1]])
            
            method = msg.get('method', 'Unknown')
            color = SIP_METHOD_COLORS.get(method, '#78909C')
            
            line = QGraphicsLineItem(src_x, y, dst_x, y)
            line.setPen(QPen(QColor(color), 2))
            self.scene.addItem(line)
            
            if src_x < dst_x:
                points = [QPointF(dst_x-8, y-5), QPointF(dst_x, y), QPointF(dst_x-8, y+5)]
            else:
                points = [QPointF(dst_x+8, y-5), QPointF(dst_x, y), QPointF(dst_x+8, y+5)]
            arrow = QGraphicsPolygonItem(QPolygonF(points))
            arrow.setBrush(QBrush(QColor(color)))
            arrow.setPen(QPen(QColor(color)))
            self.scene.addItem(arrow)
            
            label_text = QGraphicsTextItem(method)
            label_text.setDefaultTextColor(QColor('white'))
            label_text.setFont(QFont("Arial", 7, QFont.Bold))
            
            label_bg = QGraphicsRectItem(
                (src_x + dst_x)/2 - 25, y - 18,
                50, 16
            )
            label_bg.setBrush(QBrush(QColor(color)))
            label_bg.setPen(QPen(Qt.NoPen))
            self.scene.addItem(label_bg)
            
            label_text.setPos((src_x + dst_x)/2 - label_text.boundingRect().width()/2, y - 18)
            self.scene.addItem(label_text)
            
            y += msg_spacing

    def update_display(self):
        self.stream_table.setRowCount(len(self.streams))
        
        for row, (ssrc, stream) in enumerate(self.streams.items()):
            self.stream_table.setItem(row, 0, QTableWidgetItem(str(ssrc)))
            self.stream_table.setItem(row, 1, QTableWidgetItem(stream.codec['name'][:15]))
            self.stream_table.setItem(row, 2, QTableWidgetItem(str(stream.packet_count)))
            
            stream.detect_packet_loss()
            lost = QTableWidgetItem(str(stream.lost_packets))
            if stream.lost_packets > 0:
                lost.setBackground(QBrush(QColor(COLORS['danger'])))
            self.stream_table.setItem(row, 3, lost)
            
            jitter = QTableWidgetItem(f"{stream.avg_jitter:.1f}")
            if stream.avg_jitter > 30:
                jitter.setBackground(QBrush(QColor(COLORS['danger'])))
            self.stream_table.setItem(row, 4, jitter)
            
            mos = stream.calculate_mos()
            mos_item = QTableWidgetItem(f"{mos:.2f}")
            if mos < 3.0:
                mos_item.setBackground(QBrush(QColor(COLORS['danger'])))
            elif mos < 4.0:
                mos_item.setBackground(QBrush(QColor(COLORS['warning'])))
            else:
                mos_item.setBackground(QBrush(QColor(COLORS['success'])))
            self.stream_table.setItem(row, 5, mos_item)
            
            duration = stream.end_time - stream.start_time if stream.start_time and stream.end_time else 0
            self.stream_table.setItem(row, 6, QTableWidgetItem(f"{duration:.1f}s"))
            
            action_w = QWidget()
            action_l = QHBoxLayout(action_w)
            action_l.setContentsMargins(2, 2, 2, 2)
            action_l.setSpacing(2)
            
            export = QPushButton("💾")
            export.setMaximumWidth(30)
            export.setStyleSheet(f"QPushButton {{ background-color: {COLORS['success']}; color: white; border: none; padding: 2px; font-size: 10pt; }}")
            export.clicked.connect(lambda _, s=ssrc: self.export_stream(s))
            action_l.addWidget(export)
            
            play = QPushButton("▶️")
            play.setMaximumWidth(30)
            play.setStyleSheet(f"QPushButton {{ background-color: {COLORS['primary']}; color: white; border: none; padding: 2px; font-size: 10pt; }}")
            play.clicked.connect(lambda _, s=ssrc: self.play_stream(s))
            action_l.addWidget(play)
            
            self.stream_table.setCellWidget(row, 7, action_w)
        
        self.stream_table.resizeColumnsToContents()
    
    def export_stream(self, ssrc):
        if ssrc not in self.streams:
            return
        
        stream = self.streams[ssrc]
        # Default to PROGRAM_DATA_DIR for audio exports
        default_path = os.path.join(AUDIO_EXPORT_DIR, f'rtp_{ssrc}.wav')
        filename, _ = QFileDialog.getSaveFileName(self, 'Export Audio', default_path, 'WAV Files (*.wav)')
        if filename:
            if stream.export_to_wav(filename):
                QMessageBox.information(self, 'Success', f'Audio exported to:\n{filename}')
            else:
                QMessageBox.critical(self, 'Error', 'Export failed')
    
    def play_stream(self, ssrc):
        if ssrc not in self.streams:
            return
        
        stream = self.streams[ssrc]
        # Use PROGRAM_DATA_DIR for temp files
        temp_dir = AUDIO_EXPORT_DIR
        os.makedirs(temp_dir, exist_ok=True)
        temp = os.path.join(temp_dir, f'temp_{ssrc}.wav')
        if stream.export_to_wav(temp):
            try:
                if sys.platform == "win32":
                    os.startfile(temp)
                elif sys.platform == "darwin":
                    subprocess.call(['open', temp])
                else:
                    subprocess.call(['xdg-open', temp])
            except Exception as e:
                QMessageBox.warning(self, 'Error', f'Could not play audio: {e}')
    
    def export_all(self):
        if not self.streams:
            QMessageBox.information(self, 'No Streams', 'No RTP streams to export')
            return
        
        # Default to PROGRAM_DATA_DIR for export directory
        dir_path = QFileDialog.getExistingDirectory(self, 'Select Export Directory', AUDIO_EXPORT_DIR)
        if dir_path:
            count = 0
            for ssrc, stream in self.streams.items():
                if stream.export_to_wav(os.path.join(dir_path, f'rtp_{ssrc}.wav')):
                    count += 1
            QMessageBox.information(self, 'Export Complete', f'Exported {count} of {len(self.streams)} streams')
    
    def clear_streams(self):
        self.streams.clear()
        self.stream_table.setRowCount(0)
    
    def clear_streams_with_confirm(self):
        if QMessageBox.question(self, 'Clear Streams', 'Clear all RTP streams?') == QMessageBox.Yes:
            self.clear_streams()

    def process_packet_batch(self):
        with self.packet_batch_lock:
            if not self.packet_batch:
                return
            batch = self.packet_batch[:]
            self.packet_batch.clear()
        
        for pkt in batch:
            self.packet_table.add_packet(pkt)
        
        self._update_data_rate(sum(pkt.get('length', 0) for pkt in batch))
        
        if len(batch) > 0:

    def clear_all(self):
        """Clear all data with comprehensive guards"""
        try:
            if self._closing:
                return
            
            # Check for active processes
            if self._is_loading:
                QMessageBox.warning(self, 'Busy', 'Please wait for loading to complete')
                return
            
            if self.capturing:
                reply = QMessageBox.question(
                    self, 'Capture Active',
                    'Capture is currently running. Clear all data anyway?',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.No:
                    return
            
            reply = QMessageBox.question(
                self, 'Clear All',
                'Clear all captured data?',
                QMessageBox.Yes | QMessageBox.No
            )
            if reply != QMessageBox.Yes:
                return
            
            # Clear data structures safely
            try:
                with self.packet_batch_lock:
                    self.packet_batch.clear()
                with self.voip_queue_lock:
                    self.voip_parse_queue.clear()
                
                if self.packet_table:
                    self.packet_table.clear_all()
                self.sip_calls.clear()
                self.rtp_streams.clear()

                    self.call_combo.clear()

                    self.rtp_widget.clear_streams()
                
                self.update_status()
            except Exception as e:
                LOGGER.error("Clear operation failed: %s", e, exc_info=True)
                QMessageBox.warning(self, 'Warning', f'Some data may not have been cleared:\n{str(e)}')
                
        except Exception as e:
            LOGGER.error("Clear all failed: %s", e, exc_info=True)
            try:
                QMessageBox.critical(self, 'Error', f'Failed to clear data:\n{str(e)}')

                pass

    def safe_ui_operation(self, func, *args, **kwargs):
        """Safely execute UI operation on main thread with exception handling"""
        try:
            if self._closing:
                return
            # Check if we're on the main thread
            if QThread.currentThread() == self.thread():
                func(*args, **kwargs)
            else:
                QTimer.singleShot(0, lambda f=func, a=args, kw=kwargs: self._safe_ui_exec(f, a, kw))
        except Exception as e:
            LOGGER.error("UI operation failed: %s", e, exc_info=True)
    
    def _safe_ui_exec(self, func, args, kwargs):
        """Execute UI operation on main thread"""
        try:
            if not self._closing:
                func(*args, **kwargs)
        except Exception as e:
            LOGGER.error("UI execution failed: %s", e, exc_info=True)
    
    def update_status(self):

        try:
            with self._status_update_lock:
                if self._closing:

                packet_count = len(self.packet_table.packets) if self.packet_table else 0
                call_count = len(self.sip_calls)

    def open_pcap(self):
        """THREAD-SAFE PCAP loading with comprehensive guards"""
        try:
            if self._closing:
                return

            if self._is_loading:
                QMessageBox.warning(self, 'Busy', 'Already loading a file. Please wait...')
                return

            if self.pcap_loader and self.pcap_loader.is_running():
                QMessageBox.warning(self, 'Busy', 'Already loading a file. Please wait...')
                return

            if self.capturing:
                reply = QMessageBox.question(
                    self, 'Capture Active',
                    'Capture is currently running. Stop capture and load file?',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.Yes:
                    self.stop_capture()
                else:
                    return

            # Default to PROGRAM_DATA_DIR for opening files
            filename, _ = QFileDialog.getOpenFileName(
                self, 'Open PCAP File',
                PROGRAM_DATA_DIR, 'PCAP Files (*.pcap *.pcapng *.cap)'
            )
            if not filename:
                return

            if self.packet_table and self.packet_table.packets:
                reply = QMessageBox.question(
                    self, 'Clear Current Data?',
                    'Clear current packets before loading?',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply != QMessageBox.Yes:
                    return

            self.lock_ui()
            try:
                with self.packet_batch_lock:
                    self.packet_batch.clear()
                with self.voip_queue_lock:
                    self.voip_parse_queue.clear()

                if self.packet_table:
                    self.packet_table.clear_all()
                self.sip_calls.clear()
                self.rtp_streams.clear()
                if hasattr(self, 'call_combo') and self.call_combo:
                    self.call_combo.clear()
                if hasattr(self, 'rtp_widget') and self.rtp_widget:
                    self.rtp_widget.clear_streams()

                progress = QProgressDialog("Loading PCAP file...", "Cancel", 0, 100, self)
                progress.setWindowTitle("Loading")
                progress.setWindowModality(Qt.WindowModal)
                progress.setMinimumDuration(0)

                def cancel_load():
                    if self.pcap_loader:
                        self.pcap_loader.stop()
                        self.pcap_loader.wait(1000)
                        self.pcap_loader = None
                    self.unlock_ui()

                progress.canceled.connect(cancel_load)
                progress.show()

                self.pcap_loader = PCAPLoader(filename, self.config.get('max_packets', MAX_PACKETS))
                self.pcap_loader.packet_batch_loaded.connect(self.on_pcap_batch_loaded)
                self.pcap_loader.loading_progress.connect(
                    lambda curr, total: progress.setValue(int((curr / total * 100)) if total > 0 else 0)
                )
                self.pcap_loader.loading_complete.connect(
                    lambda count: self.on_pcap_load_complete(progress, filename, count)
                )
                self.pcap_loader.loading_error.connect(
                    lambda err: self.on_pcap_load_error(progress, err)
                )

                self.pcap_loader.start()
            except Exception as load_error:
                LOGGER.error("PCAP load initiation failed: %s", load_error, exc_info=True)
                QMessageBox.critical(self, 'Load Error', f'Failed to start load:\n{str(load_error)}')
                self.unlock_ui()
        except Exception as e:
            LOGGER.error("Open PCAP failed: %s", e, exc_info=True)
            self.unlock_ui()
            try:
                QMessageBox.critical(self, 'Error', f'Failed to open PCAP:\n{str(e)}')
            except:
                pass
    
    def on_pcap_batch_loaded(self, packet_batch):
        """Safely process packet batch from PCAP loader thread"""
        try:
            if not packet_batch or not isinstance(packet_batch, list):
                return
                
            for packet_info in packet_batch:
                try:
                    if packet_info and isinstance(packet_info, dict):
                        if self.packet_table:
                            self.packet_table.add_packet(packet_info)
                        
                        if self.is_likely_voip(packet_info):
                            with self.voip_queue_lock:
                                self.voip_parse_queue.append(packet_info)
                except Exception as packet_error:
                    # Skip individual packet errors to prevent crash
                    LOGGER.debug("Error processing packet in batch: %s", packet_error)
                    continue
        except Exception as batch_error:
            # Prevent crash from batch processing errors
            LOGGER.error("Error processing packet batch: %s", batch_error, exc_info=True)
    
    def process_voip_queue(self):
        with self.voip_queue_lock:
            if not self.voip_parse_queue:
                return
            
            batch_size = 50
            batch = self.voip_parse_queue[:batch_size]
            self.voip_parse_queue = self.voip_parse_queue[batch_size:]
        
        for packet_info in batch:
            try:
                self.parse_voip_from_packet_info_fast(packet_info)
            except:
                pass
    
    def is_likely_voip(self, packet_info):
        protocol = packet_info.get('protocol', '')
        sport = packet_info.get('sport', 0)
        dport = packet_info.get('dport', 0)
        
        if protocol in ['SIP', 'RTP']:
            return True
        
        if sport in [5060, 5061] or dport in [5060, 5061]:
            return True
        
        if 10000 <= sport <= 20000 or 10000 <= dport <= 20000:
            return True
        
        return False
    
    def parse_voip_from_packet_info_fast(self, packet_info):
        if not packet_info.get('raw_packet'):
            return
        
        protocol = packet_info.get('protocol', '')
        sport = packet_info.get('sport', 0)
        dport = packet_info.get('dport', 0)
        
        try:
            from scapy.all import Ether, Raw
            packet = Ether(packet_info['raw_packet'])
            
            if not packet.haslayer(Raw):
                return
            
            raw_data = bytes(packet[Raw].load)
            
            if protocol == 'SIP' or sport in [5060, 5061] or dport in [5060, 5061]:
                if b'SIP/' not in raw_data and b'INVITE' not in raw_data:
                    return
                
                try:
                    sip_data = raw_data.decode('utf-8', errors='ignore')
                    
                    if 'SIP/' not in sip_data:
                        return
                    
                    lines = sip_data.split('\r\n')
                    if not lines or len(lines) < 2:
                        return
                    
                    sip_info = {
                        'timestamp': time.time(),
                        'src': packet_info['src'],
                        'dst': packet_info['dst'],
                        'sport': sport,
                        'dport': dport,
                        'raw': sip_data[:500],
                        'headers': {}
                    }
                    
                    first_line = lines[0]
                    if first_line.startswith('SIP/'):
                        parts = first_line.split(' ', 2)
                        sip_info['type'] = 'response'
                        sip_info['status_code'] = parts[1] if len(parts) > 1 else 'Unknown'
                        sip_info['method'] = sip_info['status_code']
                    else:
                        parts = first_line.split(' ')
                        sip_info['type'] = 'request'
                        sip_info['method'] = parts[0] if parts else 'Unknown'
                    
                    for line in lines[1:20]:
                        if ':' in line:
                            key, value = line.split(':', 1)
                            key = key.strip()
                            if key in ['Call-ID', 'From', 'To']:
                                sip_info['headers'][key] = value.strip()
                    
                    sip_info['call_id'] = sip_info['headers'].get('Call-ID', '')
                    sip_info['from'] = sip_info['headers'].get('From', '')
                    sip_info['to'] = sip_info['headers'].get('To', '')
                    
                    if sip_info['call_id']:
                        self.on_sip(sip_info)
                except:
                    pass
            
            elif protocol == 'RTP' or (10000 <= sport <= 20000) or (10000 <= dport <= 20000):
                if len(raw_data) < 12:
                    return
                
                version = (raw_data[0] >> 6) & 0x03
                if version != 2:
                    return
                
                payload_type = raw_data[1] & 0x7F
                if payload_type > 127:
                    return
                
                try:
                    cc = raw_data[0] & 0x0F
                    sequence = struct.unpack('!H', raw_data[2:4])[0]
                    timestamp = struct.unpack('!I', raw_data[4:8])[0]
                    ssrc = struct.unpack('!I', raw_data[8:12])[0]
                    
                    header_len = 12 + (cc * 4)
                    if header_len <= len(raw_data):
                        rtp_payload = raw_data[header_len:header_len+100]
                        
                        rtp_info = {
                            'timestamp': time.time(),
                            'src': packet_info['src'],
                            'dst': packet_info['dst'],
                            'sport': sport,
                            'dport': dport,
                            'ssrc': ssrc,
                            'sequence': sequence,
                            'rtp_timestamp': timestamp,
                            'payload_type': payload_type,
                            'payload': rtp_payload
                        }
                        
                        self.on_rtp(rtp_info)
                except:
                    pass
        except:
            pass

    def save_pcap(self):
        """Save PCAP file with comprehensive guards and exception handling"""
        try:
            if self._closing:
                return
            
            # Check for active processes
            if self._is_loading:
                QMessageBox.warning(self, 'Busy', 'Please wait for loading to complete')
                return
            
            if self.capturing:
                reply = QMessageBox.question(
                    self, 'Capture Active',
                    'Capture is currently running. Save anyway?\n\nNote: Only packets captured so far will be saved.',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.No:

            filename, _ = QFileDialog.getSaveFileName(
                self, 'Save PCAP',
                default_path,
                'PCAP Files (*.pcap)'
            )
            if not filename:
                return
            
            # Save in a thread to avoid blocking UI
            def _save_operation():
                try:
                    raw_packets = []
                    packets_copy = list(self.packet_table.packets)  # Make a copy to avoid race conditions
                    
                    for pkt_info in packets_copy:
                        if 'raw_packet' in pkt_info and pkt_info['raw_packet']:
                            try:
                                pkt = Ether(pkt_info['raw_packet'])
                                raw_packets.append(pkt)
                            except Exception as e:
                                LOGGER.debug("Failed to parse packet for save: %s", e)
                                continue
                    
                    if not raw_packets:
                        self.safe_ui_operation(
                            QMessageBox.warning,
                            self, 'No Raw Data',
                            'No raw packet data available for export'
                        )
                        return
                    
                    wrpcap(filename, raw_packets)
                    self.safe_ui_operation(
                        QMessageBox.information,
                        self, 'PCAP Saved',
                        f'✅ Saved {len(raw_packets):,} packets to:\n{filename}'
                    )
                except Exception as e:

                    self.safe_ui_operation(
                        QMessageBox.critical,
                        self, 'Save Error',
                        f'Failed to save PCAP:\n{str(e)}'
                    )
            
            # Run save in background thread
            save_thread = threading.Thread(target=_save_operation, daemon=True)
            save_thread.start()
            
        except Exception as e:
            LOGGER.error("Save PCAP failed: %s", e, exc_info=True)
            try:
                QMessageBox.critical(self, 'Error', f'Failed to save PCAP:\n{str(e)}')

                pass

    def start_remote_capture_agent(self, agent):
        """Start remote capture agent - Thread-safe with proper error handling"""
        try:
            # Try enhanced capture first (Wireshark-compatible), then fallback to original
            success = False
            try:
                success = agent.start_enhanced_capture(filters=None, packet_callback=self.on_remote_packet)
            except Exception as e:
                LOGGER.error("Enhanced capture failed: %s", e, exc_info=True)
            
            if not success:
                try:
                    success = agent.start_remote_capture(filters=None, packet_callback=self.on_remote_packet)
                except Exception as e:
                    LOGGER.error("Remote capture failed: %s", e, exc_info=True)
            
            if success:
                agent.is_capturing = True
                agent.last_packet_ts = time.time()
                agent.last_status_update = agent.last_packet_ts
                if getattr(agent, 'controls', None):
                    agent.controls.get('start', None).setEnabled(False)
                    agent.controls.get('stop', None).setEnabled(True)
                self.update_remote_agent_row(agent, "Capturing", 'warning')
                QMessageBox.information(self, 'Remote Capture Started', 'Remote capture is now active.\n\nPackets will appear in the Packets tab.')
            else:
                QMessageBox.critical(self, 'Error', 'Failed to start remote capture.\n\nCheck connection and try again.')
        except Exception as e:
            LOGGER.error("Start remote capture agent failed: %s", e, exc_info=True)
            QMessageBox.critical(self, 'Error', f'Failed to start remote capture:\n{str(e)}')

    def on_remote_packet(self, packet_info, agent=None):
        """Handle packets received from remote agent - Thread-safe GUI update"""
        try:
            # Ensure packet has required fields
            if not packet_info or not isinstance(packet_info, dict):
                return
            
            # Add missing fields if not present
            if 'timestamp' not in packet_info:
                packet_info['timestamp'] = str(datetime.now())
            if 'frame_number' not in packet_info:
                packet_info['frame_number'] = len(self.packet_table.packets) + 1
            if 'raw_packet' not in packet_info:
                packet_info['raw_packet'] = b''
            
            # Process the packet (this is thread-safe as on_packet uses locks)
            self.on_packet(packet_info)
            
            # Check for VoIP
            if self.is_likely_voip(packet_info):
                with self.voip_queue_lock:
                    self.voip_parse_queue.append(packet_info)
                    
        except Exception as e:
            # Log but don't crash - thread safety is critical
            LOGGER.error("Error processing remote packet: %s", e, exc_info=True)
                
        # Update status
        self.update_status()

    def closeEvent(self, event):
        try:
            self._closing = True
            self.update_action_states()

            if self._is_loading:
                reply = QMessageBox.question(
                    self, 'Loading in Progress',
                    'File loading in progress. Stop and exit?',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.No:
                    self._closing = False
                    event.ignore()
                    return
                if self.pcap_loader:
                    try:
                        self.pcap_loader.stop()
                        self.pcap_loader.wait(2000)
                    except Exception:
                        pass
                    finally:
                        self.pcap_loader = None

            if self.capturing:
                reply = QMessageBox.question(
                    self, 'Capture Active',
                    'Capture is active. Stop and exit?',
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply == QMessageBox.No:
                    self._closing = False
                    event.ignore()
                    return
                try:
                    self.stop_capture()
                except Exception as e:
                    LOGGER.error("Error stopping capture on close: %s", e, exc_info=True)

            # Stop timers
            try:
                if hasattr(self, 'batch_timer') and self.batch_timer:
                    self.batch_timer.stop()
                if hasattr(self, 'voip_parse_timer') and self.voip_parse_timer:
                    self.voip_parse_timer.stop()
                if hasattr(self, 'remote_health_timer') and self.remote_health_timer:
                    self.remote_health_timer.stop()
            except Exception:
                pass

            # Disconnect remote agents
            try:
                for agent in self.remote_agents:
                    try:
                        agent.disconnect()
                    except Exception:
                        pass
                self.remote_agents.clear()
            except Exception:
                pass

            # Detach log handler
            try:
                if self._log_handler:
                    LOGGER.removeHandler(self._log_handler)
                    self._log_handler = None
            except Exception:
                pass

            save_config(self.config)
            event.accept()
        except Exception as e:
            LOGGER.error("Close event failed: %s", e, exc_info=True)
            event.accept()

    def update_remote_agent_health(self):
        try:
            now = time.time()
            for idx, agent in enumerate(self.remote_agents):
                agent.table_row = idx
                status_text, color_key = self._remote_agent_status(agent, now)
                self.update_remote_agent_row(agent, status_text, color_key)
        except Exception as e:
            LOGGER.error("Remote health update failed: %s", e, exc_info=True)
 
    def _remote_agent_status(self, agent, now=None):
        now = now or time.time()
        if getattr(agent, 'is_capturing', False):
            last_ts = getattr(agent, 'last_packet_ts', None)
            if last_ts is None:
                return ("Awaiting packets…", 'warning')
            gap = now - last_ts
            if gap > 10:
                return (f"Stale ({int(gap)}s)", 'danger')
            if gap > 5:
                return (f"Slow ({int(gap)}s)", 'warning')
            return ("Capturing", 'success')
        return ("Connected", 'success')

    def update_remote_agent_row(self, agent, status_text, color_key='success'):
        try:
            if agent is None or agent.table_row is None:
                return
            item = self.agent_table.item(agent.table_row, 2)
            if item is None:
                item = QTableWidgetItem()
                self.agent_table.setItem(agent.table_row, 2, item)
            item.setText(status_text)
            color = COLORS.get(color_key, COLORS['light'])
            item.setForeground(QBrush(QColor(color)))
        except Exception as e:
            LOGGER.error("Failed to update agent row: %s", e, exc_info=True)

    def _update_data_rate(self, data_size):
        try:
            window_ms = max(self.config.get('update_interval', 200), 1)
            self._data_rate_kbps = (data_size / 1024.0) / (window_ms / 1000.0)
            self._rate_window.append((time.time(), self._data_rate_kbps))
            cutoff = time.time() - 5.0
            while self._rate_window and self._rate_window[0][0] < cutoff:
                self._rate_window.popleft()
            avg_rate = sum(val for _, val in self._rate_window) / len(self._rate_window) if self._rate_window else 0.0
            self.stats_dashboard.update_stats(data_rate=avg_rate)
        except Exception as e:
            LOGGER.error("Data rate update failed: %s", e, exc_info=True)

    def update_action_states(self):
        """Enable/disable menu and toolbar actions based on current state"""
        try:
            loading = self._is_loading
            capturing = self.capturing
            closing = self._closing

            def _set_enabled(actions, enabled):
                for act in actions:
                    try:
                        if act is not None:
                            act.setEnabled(enabled)
                    except Exception:
                        continue

            general_enabled = not closing and not loading
            capture_allowed = general_enabled and not capturing
            capture_active = general_enabled and capturing

            _set_enabled(self._loading_sensitive_actions, general_enabled)
            _set_enabled(self._capture_sensitive_actions, capture_allowed)
            _set_enabled(self._capture_start_actions, capture_allowed)
            _set_enabled(self._capture_stop_actions, capture_active)
        except Exception as e:
            LOGGER.error("Failed to update action states: %s", e, exc_info=True)

    def append_log_message(self, message, level=logging.INFO):
        if not hasattr(self, 'log_output') or self._closing:
            return
        timestamp = datetime.now().strftime('%H:%M:%S')
        entry = f"{timestamp} | {message}"
        self.log_output.appendPlainText(entry)

        <h1>Spin the Wheel - 30 Items</h1>

            <input type="text" id="itemInput" placeholder="Add an item...">
            <button id="addBtn">Add Item</button>
        </div>
        
        <div class="wheel-container">
            <div class="pointer"></div>
            <div id="wheel" class="wheel"></div>
        </div>
        
        <button id="spinBtn">SPIN</button>
        
        <div class="result" id="result">Add items and spin the wheel!</div>
        
        <div class="items-list">
            <h3>Items List (<span id="itemCount">0</span>/30)</h3>
            <ul id="items"></ul>
        </div>
        
        <div class="controls">
            <button id="clearBtn" class="action-btn">Clear All</button>
            <button id="sampleBtn" class="action-btn">Add Sample Items</button>
        </div>
    </div>

    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const items = [];
            const wheel = document.getElementById('wheel');
            const itemInput = document.getElementById('itemInput');
            const addBtn = document.getElementById('addBtn');
            const spinBtn = document.getElementById('spinBtn');
            const clearBtn = document.getElementById('clearBtn');
            const sampleBtn = document.getElementById('sampleBtn');
            const itemsList = document.getElementById('items');
            const resultDiv = document.getElementById('result');
            const itemCount = document.getElementById('itemCount');
            
            let rotation = 0;
            let spinning = false;
            let currentItems = [];
            
            // Add item to the list
            addBtn.addEventListener('click', addItem);
            itemInput.addEventListener('keypress', function(e) {
                if (e.key === 'Enter') addItem();
            });
            
            function addItem() {
                if (items.length >= 30) {
                    resultDiv.textContent = 'Maximum 30 items reached!';
                    return;
                }
                
                const item = itemInput.value.trim();
                if (item && !spinning) {
                    items.push(item);
                    itemInput.value = '';
                    updateItemsList();
                    drawWheel();
                }
            }
            
            // Clear all items
            clearBtn.addEventListener('click', function() {
                if (!spinning) {
                    items.length = 0;
                    updateItemsList();
                    drawWheel();
                    resultDiv.textContent = 'All items cleared!';
                }
            });
            
            // Add sample items
            sampleBtn.addEventListener('click', function() {
                if (!spinning && items.length < 30) {
                    const samples = [
                        "Pizza", "Burger", "Sushi", "Tacos", "Pasta",
                        "Salad", "Steak", "Chicken", "Fish", "Soup",
                        "Sandwich", "Curry", "Ramen", "Dumplings", "BBQ",
                        "Seafood", "Veggies", "Fruit", "Dessert", "Coffee",
                        "Tea", "Smoothie", "Juice", "Ice Cream", "Cake",
                        "Cookies", "Wine", "Beer", "Cocktail", "Water"
                    ];
                    
                    for (let i = 0; i < samples.length && items.length < 30; i++) {
                        items.push(samples[i]);
                    }
                    
                    updateItemsList();
                    drawWheel();
                }
            });
            
            // Update the items list display
            function updateItemsList() {
                itemsList.innerHTML = '';
                items.forEach((item, index) => {
                    const li = document.createElement('li');
                    li.textContent = item;
                    
                    const deleteBtn = document.createElement('button');
                    deleteBtn.textContent = 'Delete';
                    deleteBtn.className = 'delete-btn';
                    deleteBtn.onclick = () => {
                        if (!spinning) {
                            items.splice(index, 1);
                            updateItemsList();
                            drawWheel();
                        }
                    };
                    
                    li.appendChild(deleteBtn);
                    itemsList.appendChild(li);
                });
                
                itemCount.textContent = items.length;
            }
            
            // Draw the wheel
            function drawWheel() {
                wheel.innerHTML = '';
                currentItems = [...items];
                
                if (currentItems.length === 0) {
                    const noItemsText = document.createElement('div');
                    noItemsText.style.position = 'absolute';
                    noItemsText.style.top = '50%';
                    noItemsText.style.left = '50%';
                    noItemsText.style.transform = 'translate(-50%, -50%)';
                    noItemsText.style.fontSize = '16px';
                    noItemsText.style.color = '#333';
                    noItemsText.textContent = 'Add items to begin';
                    wheel.appendChild(noItemsText);
                    return;
                }
                
                const sliceAngle = 360 / currentItems.length;
                const colors = [
                    '#6a11cb', '#2575fc', '#ff5252', '#4caf50', '#ff9800', 
                    '#9c27b0', '#2196f3', '#f44336', '#009688', '#673ab7',
                    '#3f51b5', '#03a9f4', '#00bcd4', '#cddc39', '#ffeb3b',
                    '#ffc107', '#ff5722', '#795548', '#607d8b', '#8bc34a',
                    '#e91e63', '#9e9e9e', '#00e676', '#ff4081', '#76ff03',
                    '#ff9100', '#651fff', '#1de9b6', '#d500f9', '#c6ff00'
                ];
                
                for (let i = 0; i < currentItems.length; i++) {
                    const section = document.createElement('div');
                    section.className = 'wheel-section';
                    section.style.transform = `rotate(${i * sliceAngle}deg)`;
                    section.style.background = colors[i % colors.length];
                    
                    const text = document.createElement('div');
                    text.textContent = currentItems[i];
                    text.style.transform = `rotate(${sliceAngle/2}deg)`;
                    text.style.paddingRight = '80px';
                    text.style.width = '120px';
                    text.style.fontSize = currentItems.length > 20 ? '10px' : '12px';
                    
                    section.appendChild(text);
                    wheel.appendChild(section);
                }
            }
            
            // Spin the wheel
            spinBtn.addEventListener('click', function() {
                if (currentItems.length < 2) {
                    resultDiv.textContent = 'Add at least 2 items to spin!';
                    return;
                }
                
                if (spinning) return;
                
                spinning = true;
                resultDiv.textContent = 'Spinning...';
                
                // Calculate random rotation (at least 2 full rotations plus random)
                const extraRotation = 2 * 360 + Math.random() * 360;
                const randomIndex = Math.floor(Math.random() * currentItems.length);
                const sliceAngle = 360 / currentItems.length;
                
                // Adjust rotation to land on selected item
                const targetRotation = extraRotation + (360 - (randomIndex * sliceAngle + sliceAngle / 2));
                
                rotation += targetRotation;
                
                // Apply rotation with CSS
                wheel.style.transform = `rotate(${rotation}deg)`;
                
                // Display result after animation
                setTimeout(() => {
                    resultDiv.textContent = `Landed on: ${currentItems[randomIndex]}`;
                    spinning = false;
                }, 4000);
            });
            
            // Initial draw
            drawWheel();
        });
    </script>
